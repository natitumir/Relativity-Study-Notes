\section{Topological Space}

\begin{definition}
    Let M be a set. We define a topology $ \mathcal{O}$ on M to be a collection of subsets of M such that
    \begin{enumerate}[label=(\roman*)]
        \item $ \emptyset, M \in \mathcal{O}$
        \item $ \mathcal{O}$ is closed under arbitrary union and finite intersection. i.e.
    \end{enumerate}

    \begin{align*}
        & U, V \in \mathcal{O} \implies U \cap V \in \mathcal{O} \\
        & U_\alpha \in \mathcal{O} \lera{ \alpha \in A} \implies \bigcup_{\alpha \in A} U_\alpha \in \mathcal{O} \\
    \end{align*}
\end{definition}

\begin{definition}
    A set M equipped with a topology $ \mathcal{O}$, i.e. $ \lera{ M, \mathcal{O}}$ is called a topological space.
\end{definition}

Examples of topological spaces are as follows. For any set M, the collection $ \mathcal{O} = \{ \emptyset, M \}$ is a topology. This is called the chaotic topology. The collection $ \mathcal{O} = P(M)$ (Power set of M) is called the discrete topology. Another example is $ \lera{M, \mathcal{O}}$ with $M = \{ 1,2,3 \} $ and $ \mathcal{O} = \{ \{1\}, \{2\}, \{1,2\}, \{1,2,3\}, \emptyset \} $



Consider $ \mathbb{R}^n$ where we define the ``open ball/soft ball'' of radius r around a point $ p \in \mathbb{R}^n$ as:

\begin{align*}
    & B_r(p) := \left\{ x \in \mathbb{R}^n \; \middle| \; \sum_{i=1}^{n} | x^i - p^i |^2 < r^2 \right\} \\
\end{align*}

With the help of this we can define a topology $ \mathcal{O}_{std}$ on $ \mathbb{R}^n$.

\begin{definition}
    A set $ U \in \mathcal{O}_{std}$ iff $ U \subseteq \mathbb{R}^n$ and

    \begin{align*}
        & p \in U \implies \exists r \in \mathbb{R}^+ : B_r(p) \subseteq U \\
    \end{align*}
\end{definition}

\begin{proposition}
    $ \lera{ \mathbb{R}^n, \mathcal{O}_{std}}$ is a topological space.
\end{proposition}

\begin{proof}
    $ \emptyset \in \mathcal{O}_{std}$ since $ p \in \emptyset \implies \exists r \in \mathbb{R}^+ : B_r(p) \subseteq \emptyset$ (vacuously)

    $ \mathbb{R}^n \in \mathcal{O}_{std}$ since $ p \in \mathbb{R}^n \implies \exists r = 1 \in \mathbb{R}^+ : B_r(p) \subseteq \mathbb{R}^n$

    If $ U, V \in \mathcal{O}_{std}$ then assume $ p \in U \cap V$

    \begin{align*}
        p \in U \cap V &\implies p \in U \text{ and } p \in V \\
        &\implies \exists r_1, r_2 \in \mathbb{R}^+ : B_{r_1}(p) \subseteq U \text{ and } B_{r_2}(p) \subseteq V \\
    \end{align*}

    Now, $ r := \min \lera{ r_1, r_2}$

    \begin{align*}
        & \therefore B_r(p) \subseteq U \text{ and } B_r(p) \subseteq V \\
        & \implies B_r(p) \subseteq U \cap V \\
        & \therefore U, V \in \mathcal{O}_{std} \implies \lera{ U \cap V} \in \mathcal{O}_{std} \\
    \end{align*}

    Assume $ U_\alpha \in \mathcal{O}_{std}$ $ \lera{ \alpha \in A}$

    Let $ p \in \bigcup_{\alpha \in A} U_\alpha$

    \begin{align*}
        & \implies p \in U_\beta \text{ for some } \beta \in A \\
        & \implies \exists r \in \mathbb{R}^+ : B_r(p) \subseteq U_\beta \\
        & \implies \exists r \in \mathbb{R}^+ : B_r(p) \subseteq \bigcup_{\alpha \in A} U_\alpha \\
        & \therefore U_\alpha \in \mathcal{O}_{std} \lera{ \alpha \in A} \implies \lera{ \bigcup_{\alpha \in A} U_\alpha} \in \mathcal{O}_{std} \\
    \end{align*}
\end{proof}

So, in short, we can say that $ \lera{ \mathbb{R}^n, \mathcal{O}_{std}}$ is a topological space.

\subsection*{Basis}

The basic idea is to have a collection of subsets from which we can take unions and construct any set $ U \in \mathcal{O}$.

\begin{definition}
    A collection $ \mathcal{B}$ of open sets is a basis for the topology $ \mathcal{O}$ if given any set $ U \in \mathcal{O}$ and a point $ p \in U$, there is an open set $ B \in \mathcal{B}$ such that $ p \in B \subseteq U$.
\end{definition}

An element of the basis $ \mathcal{B}$ is called a basic open set.

\begin{theorem}
    A collection of open sets $ \mathcal{B}$ is a basis of $ \mathcal{O}$ iff any open set U is a union of sets of $ \mathcal{B}$.
\end{theorem}

\begin{proof}
    $ \lera{ \implies}$ We assume that $ \mathcal{B}$ is a basis for $ \mathcal{O}$. So, if $ U \in \mathcal{O}$, then for $ p \in U$, $ \exists B_p \in \mathcal{B}$ such that $ p \in B_p \subseteq U$.

    \begin{align*}
        & \therefore U = \bigcup_{p \in U} B_p \\
    \end{align*}

    $ \lera{ \impliedby}$ Since any open set U is a union of sets of $ \mathcal{B}$, for any point $ p \in U$, we have a set $ B \in \mathcal{B}$ such that $ p \in B \subseteq U$.
\end{proof}

$ \mathcal{B}_{rat}$ is the set of all open balls $ B_r(p)$ where $ r \in \mathbb{Q}^+$ and $ p \in \mathbb{Q}^n$, i.e. rational centers and rational radius.

\begin{theorem}
    The set $ \mathcal{B}_{rat}$ is a basis for $ \mathcal{O}_{std}$.
\end{theorem}

\begin{proof}
    Take $ U \in \mathcal{O}_{std}$

    Now, for any point $ p \in U$, there exists $ r \in \mathbb{Q}^+$ such that $ B_r(p) \subseteq U$.

    Now, take a rational point $ q \in B_{r/2}(p)$

    \begin{center}
    \begin{tikzpicture}[scale=1.0]
        \draw (0,0) circle (1.5);
        \draw[dashed] (0,0) circle (0.75);
        \fill (0,0) circle (1.5pt) node[above left] {$ p$};
        \fill (0.35,-0.3) circle (1.5pt) node[above right] {$ q$};
    \end{tikzpicture}
    \end{center}

    So,

    \begin{align*}
        & d(p,q) < \frac{r}{2} \\
    \end{align*}

    Consider the ball $ B_{r/2}(q)$

    \begin{align*}
        & p \in B_{r/2}(q) \numberthis \label{p_in_half_ball} \\
    \end{align*}

    Let, $ x \in B_{r/2}(q)$

    \begin{align*}
        & \implies d(x,q) < \frac{r}{2} \\
    \end{align*}

    Now,

    \begin{align*}
        & d(x,p) \leq d(x,q) + d(q,p) < \frac{r}{2} + \frac{r}{2} = r \\
        & \implies d(x,p) < r \implies x \in B_r(p) \\
        & \therefore p \in B_{r/2}(q) \subseteq B_r(p) \subseteq U \\
    \end{align*}
\end{proof}


So, we can say that $ \mathcal{B}_{rat}$ is a basis for $ \mathcal{O}_{std}$.

There is a curious thing to note here. The set $ \mathcal{B}_{rat}$ of all the rational balls (balls with rational center and rational radii) is a countable set. Since each ball can be indexed by $ \mathbb{Q}^n$ (a countable set) and $ \mathbb{Q}^+$ (another countable set), we have a countable infinity of rational balls.

\begin{definition}
    A topological space $ \lera{ M, \mathcal{O}}$ is called second countable iff $ \mathcal{O}$ has a countable basis.
\end{definition}

\section{Hausdorff Topological Space}

\begin{definition}
    $ \lera{ T, \mathcal{O}_T}$ is called a Hausdorff top. space :=

    \begin{align*}
        \text{iff } \forall p, q \in T \text{ and } p \neq q & \\
        \implies \exists U_p \in \mathcal{O}_T \text{ and } \exists U_q \in \mathcal{O}_T, \; p \in U_p \; \& \; q \in U_q & \\
        \text{and } U_p \cap U_q = \emptyset & \\
    \end{align*}
\end{definition}

In short, for every pair of points, we have disjoint open sets that contain those points.

\begin{definition}
    Let, $ \lera{ M, \mathcal{O}_M}$ and $ \lera{ N, \mathcal{O}_N}$ be top. sp. A map $ f: M \longrightarrow N$ is called \fbox{continuous} with respect to $ \mathcal{O}_M$ and $ \mathcal{O}_N$ iff

    \begin{align*}
        & V \in \mathcal{O}_N \implies \text{preim}_f(V) \in \mathcal{O}_M \\
    \end{align*}

    i.e. preimages of open sets are open.
\end{definition}

We have defined, for $ f: M \longrightarrow N$ and $ V \subseteq N$

\begin{align*}
    & \text{preim}_f(V) = \{ m \in M \mid f(m) \in V \} \\
\end{align*}

\begin{proposition}
    Let, $ \lera{ M, \mathcal{O}_M}$, $ \lera{ N, \mathcal{O}_N}$, $ \lera{ P, \mathcal{O}_P}$ be three top. sp.

    $ f: M \longrightarrow N$, $ g: N \longrightarrow P$ are continuous maps.

    The map $ h = g \circ f$ is also continuous w.r.t. $ \mathcal{O}_M$ \& $ \mathcal{O}_P$.
\end{proposition}

\begin{proof}
    Let, $ V \in \mathcal{O}_P$

    Now,

    \begin{align*}
        \text{preim}_h(V) &= \{ m \in M \mid h(m) \in V \} \\
        &= \{ m \in M \mid \lera{ g \circ f}(m) \in V \} \\
        &= \{ m \in M \mid f(m) \in \text{preim}_g(V) \} \\
        &= \text{preim}_f \lera{ \text{preim}_g(V)} \\
        &\in \mathcal{O}_M \\
    \end{align*}
\end{proof}

\section{Inheriting a topology}

Let $ \lera{ T, \mathcal{O}}$ be a top. sp. and $ S \subseteq T$.

\begin{align*}
    & \mathcal{O}|_S := \{ U \cap S \mid U \in \mathcal{O} \} \\
\end{align*}

$ \lera{ S, \mathcal{O}|_S}$ is a top. sp.

\begin{proof}
    \begin{align*}
        & \emptyset = \emptyset \cap S \text{ and } \emptyset \in \mathcal{O} \implies \emptyset \in \mathcal{O}|_S \\
        & S = S \cap M \text{ and } M \in \mathcal{O} \implies S \in \mathcal{O}|_S \\
    \end{align*}

    Let, $ S_1$ and $ S_2 \in \mathcal{O}|_S$

    \begin{align*}
        & \implies \exists U_1, U_2 \in \mathcal{O} \text{ such that } S_1 = U_1 \cap S, \; S_2 = U_2 \cap S \\
        & \implies U_1 \cap U_2 \in \mathcal{O} \implies \lera{ U_1 \cap U_2} \cap S \in \mathcal{O}|_S \\
        & \implies \lera{ U_1 \cap S} \cap \lera{ U_2 \cap S} \in \mathcal{O}|_S \implies S_1 \cap S_2 \in \mathcal{O}|_S \\
    \end{align*}

    Let, $ \alpha \in I$ and $ S_\alpha \in \mathcal{O}|_S$ $ \forall \alpha \in I$

    \begin{align*}
        & \implies \forall \alpha \; \exists U_\alpha \in \mathcal{O} \text{ such that } S_\alpha = U_\alpha \cap S \\
        & \implies \bigcup_{\alpha \in I} U_\alpha \in \mathcal{O} \implies \lera{ \bigcup_{\alpha \in I} U_\alpha} \cap S \in \mathcal{O}|_S \\
        & \implies \bigcup_{\alpha \in I} \lera{ U_\alpha \cap S} \in \mathcal{O}|_S \implies \bigcup_{\alpha \in I} S_\alpha \in \mathcal{O}|_S \\
    \end{align*}
\end{proof}

\section{Top. Manifold}

A Hausdorff, second countable top. space $ \lera{ M, \mathcal{O}}$ is called a d-dimensional topological manifold if for every point $ p \in M$, there is an open set U such that $ p \in U$ and there is a homeomorphism $ x: U \longrightarrow x(U) \subseteq \mathbb{R}^d$

where $ x(U) \in \mathcal{O}_{std}$

Note:- The map x is bi-continuous w.r.t. the top. spaces $ \lera{ U, \mathcal{O}|_U}$ and $ \lera{ x(U), \mathcal{O}_{std}|_{x(U)}}$

\begin{definition}
    Chart :=

    Let $ \lera{ M, \mathcal{O}}$ be a d-dim top. mfd. Then the pair $ \lera{ U, x}$ where $ U \in \mathcal{O}$ and x is a homeomorphism $ x: U \longrightarrow x(U) \subseteq \mathbb{R}^d$, is called a chart of the manifold.
\end{definition}

\begin{definition}
    Atlas :=

    An atlas of a manifold M is a collection of charts

    \begin{align*}
        & \mathcal{A} := \{ \lera{ U_\alpha, x_\alpha} \mid \alpha \in A \} \text{ such that} \\
        & \bigcup_{\alpha \in A} U_\alpha = M \\
    \end{align*}
\end{definition}

\section{Compatibility of charts}

Two charts $ \lera{ U,x}$ and $ \lera{ V,y}$ are $ \mathcal{C}$-compatible if either $ U \cap V = \emptyset$ or the maps

\begin{align*}
    & x \circ y^{-1} : y \lera{ U \cap V} \longrightarrow x \lera{ U \cap V} \\
    & y \circ x^{-1} : x \lera{ U \cap V} \longrightarrow y \lera{ U \cap V} \\
\end{align*}

is undergraduate-$ \mathcal{C}$

Charts of a top. manifold are automatically $ \mathcal{C}^0$-compatible.

\begin{definition}
    $ \mathcal{C}$-Atlas :=

    An atlas $ \mathcal{A}$ of a top. manifold is called a $ \mathcal{C}$-Atlas if the charts of $ \mathcal{A}$ are $ \mathcal{C}$-compatible.
\end{definition}

\begin{definition}
    $ \mathcal{C}^k$-manifold:

    A $ \mathcal{C}^k$-manifold is a triple $ \lera{ M, \mathcal{O}, \mathcal{A}}$ where $ \lera{ N, \mathcal{O}}$ is a top. manifold and $ \mathcal{A}$ is a maximal $ \mathcal{C}^k$-atlas.
\end{definition}

\begin{definition}
    Smooth-manifold:

    A $ \mathcal{C}^\infty$-manifold is called a smooth manifold.
\end{definition}

From now on we are going to explicitly consider smooth manifolds.

\section{Maps between manifolds}

$ \phi: M \longrightarrow N$ is a map from M to N.

$ \phi$ will be a smooth map between M and N if the following holds:—

Let, $ p \in U \subseteq M$, $ \phi(p) \in V \subseteq N$, $ \lera{ U,x}$ and $ \lera{ V,y}$ both be charts.

\section{Algebra of Forms}








At a point p in a d-dimensional manifold M, a special class of tensors is very important and deserves a separate section of discussion. They are the totally antisymmetric (0,n)-tensors. To define them properly, let me first define an operator called the ``Antisymmetrization Operator''. For a (0,n)-tensor T at a point p in the manifold, and for a set of n linearly independent vectors $ \{ \overset{1}{X}, \overset{2}{X},...,\overset{n}{X}\}$ the action of the antisymmetrization operator $ \hat{A}$ on the tensor T is defined as,

\begin{align*}
    \hat{A}(T)\lera{\overset{1}{X}, \overset{2}{X},...,\overset{n}{X}} &= \frac{1}{n!} \sum_{ \pi} \delta_\pi \; T \lera{\overset{ \pi(1)}{X}, \overset{\pi(2)}{X},...,\overset{\pi(n)}{X}} \\
\end{align*}

where $ \pi$ denotes the different permutations of $ \{ 1,2,...,n \}$ and $ \tt{ \delta}{}{ \pi}$ is the sign of the permutation. 


The definition of $ \hat{A}$ means, $ \tt{\hat{A} \lera{T}}{}{abc} = \tt{T}{}{\left[abc\right]}$, etc. 

An n-form is a (0,n)-tensor which is the same as its antisymmetrized version. So, if $ \hat{A} \lera{T}= T$ then T is an n-form at p. As such the components of T in any chart are totally antisymmetric, i.e. antisymmetric under the exchange of any pair of indices. $ \tt{T}{}{abc}= \tt{T}{}{[abc]}$.

It is obvious that there can not be an n-form where $ n > d$. In that case, at least two of the indices must be the same. For example if $ d=3 $, then a 4-form must have indices like $ \tt{T}{}{1233}$. However, being totally antisymmetric, $ \tt{T}{}{1233}$ must be equal to $ -\tt{T}{}{1233}$ and that is possible only if $ \tt{T}{}{1233}$ is 0. 

For a component of an n-form to be non-zero, all the indices must be different. So, for example, for a 2-form A in $ d= 3$, $ \tt{A}{}{11}= \tt{A}{}{22} = \tt{A}{}{33}= 0$, however, $ \tt{A}{}{12}, \tt{A}{}{13}, \tt{A}{}{23}$ can be non-zero. 

We define a scalar as a 0-form. The space of all n-forms at the point p is denoted by $ \Omega^n_p \lera{M}$. The space of all n-form fields is denoted by $ \Omega^n(M)$. An n-form field is also called a ``differential n-form''. 

















\subsection{Wedge Product}
Given a p-form A and a q-form B at the point p, we can make a $ (p+q)$-form
through an operation called wedge product. It is denoted by the symbol $ \wedge$. Let us define this in stages. First, for a set of n 1-forms $ \{\overset{1}{ \eta},..., \overset{n}{ \eta}\}$ their wedge product is defined as, 
\begin{align*}
    \overset{1}{ \eta}  \wedge ...\wedge \overset{n}{ \eta} = n! \; \overset{[1}{ \eta} \wedge ... \wedge \overset{n]}{ \eta} \\
\end{align*}

Suppose that at point p, we have a basis of $ T_p^* M$, $ \left\{E^1,...,E^d\right\}$. The wedge product between any n of the basis elements is given by, 

\begin{align*}
    \tt{E}{a_1}{} \wedge ...\wedge E^{a_n} = n! \; E^{[a_1} \otimes...\otimes E^{a_n]} \numberthis \label{basiswedge} \\
\end{align*}


This, in turn, gives us a new way to express any $ \omega \in \Omega^n_p(M)$ wrt the basis $ \{E^i\}$. 


 
\begin{align*}
    \omega &= \omega_{a_1...a_n} E^{a_1} \otimes ... \otimes E^{a_n} \\
 &= \omega_{a_1...a_n} E^{[a_1} \otimes ... \otimes E^{a_n]} \\
 & \text{The indices of $ \omega$ are all totally antisymmetric.} \\
 &= \frac{1}{n!} \omega_{a_1...a_n} \; n! \; E^{[a_1} \otimes ... \otimes E^{a_n]} \\
  & \text{So, using \cref{basiswedge}} \\
   \omega &= \frac{1}{n!}  \omega_{a_1...a_n}  E^{a_1} \wedge ... \wedge E^{a_n} \\
\end{align*}

People also write it a bit differently sometimes, summing over only increasing indices or decreasing indices. \textcolor{blue}{finish this thought.}


So now, given a p-form A and a q-form B, what are the components of $ A \wedge B$ with respect to the basis $ \left\{E^1,...,E^d\right\}$? 

\begin{align*}
    A \wedge B &= \frac{1}{p!} \frac{1}{q!} \tt{A}{}{a_1 ... a_{p}} \tt{B}{}{b_1 ... b_q} \tt{E}{a_1}{} \wedge ... \wedge \tt{E}{a_p}{} \wedge \tt{E}{b_1}{} \wedge ... \tt{E}{b_q}{} \\
    &= \frac{ \lera{p+q}!}{p! q!}  \tt{A}{}{a_1 ... a_{p}} \tt{B}{}{b_1 ... b_q} \tt{E}{[a_1}{} \otimes ... \otimes \tt{E}{b_q]}{} \\
    &= \frac{ \lera{p+q}!}{p! q!}  \tt{A}{}{[a_1 ... a_{p}} \tt{B}{}{b_1 ... b_q]} \tt{E}{a_1}{} \otimes ... \otimes \tt{E}{b_q}{} \\
\end{align*}


So,

\begin{align*}
    \lera{A \wedge B}_{[a...bc...d]} &= \frac{ \lera{p+q}!}{p! q!} \tt{A}{}{[a...b} \tt{B}{}{c...d]} \numberthis \label{wedge_component}\\
\end{align*}

For two 1-forms $ \omega$ and $ \sigma$ we know from the definition of the wedge product that $ \omega \wedge \sigma = \omega \otimes \sigma - \sigma \otimes \omega$. This also means $ \omega \wedge \sigma = - \sigma \wedge \omega $. The rule is, any time we swap two 1-forms, there is a minus sign. Remembering this rule will lead us to a graded commutation rule for the wedge product of a p-form with a q-form. It is called graded because the anti-commutation rule depends on the degree of the forms involved. Here is the rule (assuming that $ \omega, \sigma$ are p- and q- forms, respectively). 

\begin{align*}
     \sigma \wedge \omega &= \lera{ -1}^{pq} \omega \wedge \sigma \numberthis \label{gradedcommutation} \\
\end{align*}

To see this, let's write the forms in a basis. So, $ \omega = \frac{1}{p!} \omega_{a...b}  \tt{E}{a}{} \wedge ... \wedge \tt{E}{b}{}$ and $ \sigma = \frac{1}{q!} \sigma_{c...d}  \tt{E}{c}{} \wedge ... \wedge \tt{E}{d}{} $. Now, $ \sigma \wedge \omega = \frac{1}{p! q! } \tt{ \sigma}{}{a...b} \tt{ \omega}{}{c...d} \tt{E}{a}{} \wedge...\wedge \tt{E}{b}{} \wedge \tt{E}{c}{} \wedge ... \wedge \tt{E}{d}{}$. To go to $ \omega \wedge \sigma$, the trick is to take the basis vectors from the second set $ E^{c},...,E^{d}$ one by one and take them p steps leftward. Each time I do this, I get a $ \lera{-1}^p$, and I have to do this for q members of the second set, so all in all, I have $ \lera{-1}^{pq}$. And in the end, the basis vectors will look like, $  \lera{ -1}^{pq}\tt{E}{c}{} \wedge...\wedge \tt{E}{d}{} \wedge \tt{E}{a}{} \wedge ... \wedge \tt{E}{b}{}$. So, 

\begin{align*}
    \sigma \wedge \omega &=  \frac{1}{p! q! } \tt{ \sigma}{}{a...b} \tt{ \omega}{}{c...d} \tt{E}{a}{} \wedge...\wedge \tt{E}{b}{} \wedge \tt{E}{c}{} \wedge ... \wedge \tt{E}{d}{}\\
    &= \lera{-1}^{pq}\frac{1}{p! q! } \tt{ \sigma}{}{a...b} \tt{ \omega}{}{c...d} \tt{E}{c}{} \wedge...\wedge \tt{E}{d}{} \wedge \tt{E}{a}{} \wedge ... \wedge \tt{E}{b}{}\\
    &= \lera{-1}^{pq}\frac{1}{p! q! }  \tt{ \omega}{}{c...d}\tt{ \sigma}{}{a...b} \tt{E}{c}{} \wedge...\wedge \tt{E}{d}{} \wedge \tt{E}{a}{} \wedge ... \wedge \tt{E}{b}{}\\
    &= \lera{ -1}^{pq} \omega \wedge \sigma \\
\end{align*}





















\subsection{Exterior Derivative}

I will define this using coordinate basis. For derivatives, we need n-form fields as we can not do a derivation to a quantity which is defined at a point p. So we will work in a chart $ \lera{U,x}$. The exterior derivative of a 0-form, i.e. a function f is defined as, $ df := \partial_a f dx^a $. Notice that the exterior derivative of a zero form gives a 1-form. This is a general feature of exterior derivatives as we will see below that the exterior derivative of a p-form is a $ \lera{p+1}$-form. Now, given how a function's exterior derivative is taken, we define the exterior derivative of a p-form $ A = \frac{1}{p!} \tt{A}{}{a_1 ...a_p} dx^{a_1} \wedge ... \wedge dx^{a_p}$ to be, 

\begin{align*}
    dA := \frac{1}{p!} d(A_{a_1...a_p}) \wedge \tt{dx}{a_1}{} \wedge ...\wedge \tt{dx}{a_p}{} \numberthis \label{exterior_derivative_defined} \\
\end{align*}


To see the components of dA, we expand the RHS of \cref{exterior_derivative_defined}. $ dA = \frac{1}{p!} \partial_m A_{a_1...a_p} dx^m \wedge dx^{a_1} \wedge ...dx^{a_p} = \frac{ \lera{p+1}!}{p!} \partial_m \tt{A}{}{a_1...a_p} dx^{[m} \otimes dx^{a_1} \otimes...\otimes dx^{a_p]} = \lera{p+1}\partial_{[m} \tt{A}{}{a_1...a_p]} dx^{m} \otimes dx^{a_1} \otimes...\otimes dx^{a_p}$ . So the $ \tt{dA}{}{ma_1...a_p}$ component of $ dA$ is given by $ \tt{dA}{}{ma_1...a_p}= \lera{p+1} \partial_{[m} \tt{A}{}{a_1...a_p]} $ .


We reiterate here that the RHS of \cref{exterior_derivative_defined} is a tensor by definition (exterior derivative of a function is a 1-form and wedge product of differential 1-forms is also a differential form). It is quite fascinating that a fully covariant notion of derivative can be cooked up from only coordinate-based partial derivatives.

\section{Non-coordinate Basis and Tetrad Formalism}

For non-coordinate basis, I will use Greek indices and for coordinate basis I will use Latin indices. Let us first define the non-coordinate bases. At any point p in the manifold, we can always find a set of four vectors $ \{ E_{\mu}\}$ that follow the relation, 

\begin{align*}
    g \lera{ \tt{E}{}{ \mu}, \tt{E}{}{ \nu}} &= \tt{ \eta}{}{ \mu \nu } \numberthis \label{tetrad}
\end{align*}

It is not necessary that the inner product equals $ \tt{ \eta}{}{ \mu \nu}$. However, we will make this choice in the following discussion. 

Since $ \tt{E}{ \mu}{}$ is a covector, it can be expanded in terms of the coordinate basis, 
\begin{align*}
    \tt{E}{ \mu}{} &= \tt{E}{ \mu} {a} dx^ {a} \numberthis \label{dummy3}
\end{align*}

Now, E can be thought of as a matrix. Our convention is as usual, upper index is for row, lower index is for column. And, for any matrix A, $ \tt{ \lera{A^{-1}}}{i}{j}$ is denoted by $A_j^{\;\;i}$. Based on this, $ dx^a$ can be expanded as, 

\begin{align*}
    dx^{a} &= E_{ \mu}^{\;\;a} \tt{E}{ \mu}{} \numberthis \label{dummy4}
\end{align*}

Now, the non-coordinate basis vectors $ \tt{E}{}{ \mu}$ are dual to $ \tt{E}{ \mu}{}$, that is, $ \tt{E}{ \mu}{} \lera{ \tt{E}{}{ \nu}}= \kron{ \mu}{ \nu}$. The basis vectors $ \tt{E}{}{ \mu}$ can also be expanded in terms of the coordinate basis vectors $ \partial_a$, as, $\tt{E}{}{ \mu}= F_{ \mu}^{ \;\; a} \partial_a $. The nice thing is that F is nothing but E! Let me show that.

\begin{align*}

    & \tt{E}{ \mu}{ a} \tt{ \lera{F^{-1}}}{a}{ \nu}= \kron{ \mu}{ \nu} \\
\end{align*}

	& g\lera{\partial_a, \partial_b} = g_{ab} \\
	\Rightarrow\; & g\lera{\tt{E}{\mu}{a} \tt{E}{}{\mu}, \tt{E}{\nu}{b} \tt{E}{}{\nu}} = g_{ab} \\
	\Rightarrow\; & \tt{E}{\mu}{a} \tt{E}{\nu}{b}\, g\lera{\tt{E}{}{\mu}, \tt{E}{}{\nu}} = g_{ab} \\
	\Rightarrow\; & \tt{E}{\mu}{a} \tt{E}{\nu}{b} \eta_{\mu\nu} = g_{ab}
\end{align*}

Now, consider the inverse metric, $\tilde{g}$. We know that, $\tilde{g}\lera{dx^a, dx^b} = g^{ab}$; where $g^{ab}$ is just the inverse of the matrix $g_{ab}$. Since $\tilde{g}$ is the inverse metric, we necessarily have $\tilde{g}^{\mu\nu} = \eta^{\mu\nu}$.

(5) $ \eta^{\mu\nu}\du{E}{\mu}{a} \du{E}{\nu}{b}  = g^{ab} $. The proof proceeds as follows,
\begin{align*}
	& \tilde{g}\lera{dx^a, dx^b} = g^{ab} \\
	\Rightarrow\; & \tilde{g}\lera{\du{E}{\mu}{a} \tt{E}{\mu}{}, \du{E}{\nu}{b} \tt{E}{\nu}{}} = g^{ab} \\
	\Rightarrow\; & \du{E}{\mu}{a} \du{E}{\nu}{b} \tilde{g}^{\mu\nu} = g^{ab} \\
	\Rightarrow\; & \du{E}{\mu}{a} \du{E}{\nu}{b} \eta^{\mu\nu} = g^{ab}
\end{align*}

(6) $g^{ab} \tt{E}{\mu}{a} \tt{E}{\nu}{b}  = \eta^{\mu\nu}$.  The proof proceeds as follows,
\begin{align*}
	& \tilde{g}\lera{\tt{E}{\mu}{}, \tt{E}{\nu}{}} = \eta^{\mu\nu} \\
	\Rightarrow\; & \tilde{g}\lera{\tt{E}{\mu}{a} dx^a, \tt{E}{\nu}{b} dx^b} = \eta^{\mu\nu} \\
	\Rightarrow\; & \tt{E}{\mu}{a} \tt{E}{\nu}{b} g^{ab} = \eta^{\mu\nu}
\end{align*}



We define $E^{\mu a}$ or $E_{\mu a}$ by the usual index raising and lowering using $ \tt{g}{}{mn}$. That is, $E^{\mu a} = g^{ab} \tt{E}{\mu}{b} $ and $ \tt{E}{}{ \mu a} = \tt{g}{}{ab} E_\mu{}^b$ etc. On the other hand, we can use $\eta_{\mu\nu}$ to raise and lower tetrad indices. For example, $\eta_{\mu\nu} \tt{E}{\nu}{a} = E_{\mu a}$ etc. This needs proofs, as we want the two notions of $ E_{ \mu a}$ (for example) to match.

\begin{align*}
	E_{\mu a} &= g_{ab} \du{E}{\mu}{b} \\
	&= \tt{E}{\alpha}{a} \tt{E}{\beta}{b} \eta_{\alpha\beta} \du{E}{\mu}{b} \\
	&= \tt{E}{\beta}{b} \du{E}{\mu}{b} \tt{E}{\alpha}{a} \eta_{\alpha\beta} \\
	&= \kron{\beta}{\mu} \eta_{\alpha\beta} \tt{E}{\alpha}{a} \\
	&= \eta_{\alpha\mu} \tt{E}{\alpha}{a} \\
	\Rightarrow E_{\mu a}&= \eta_{\mu\alpha} \tt{E}{\alpha}{a}
\end{align*}
And,

\begin{align*}
	E^{\mu a} &= g^{ba} \tt{E}{\mu}{b} \\
	&= \du{E}{\rho}{b} \du{E}{\sigma}{a} \eta^{\rho\sigma} \tt{E}{\mu}{b} \\
	&= \du{E}{\rho}{b} \tt{E}{\mu}{b} \du{E}{\sigma}{a} \eta^{\rho\sigma} \\
	&= \kron{\mu}{\rho} \du{E}{\sigma}{a} \eta^{\rho\sigma} \\
	\Rightarrow E^{\mu a} &= \eta^{\mu\sigma} \du{E}{\sigma}{a}
\end{align*}








It is nice to see how handy the tensor index gymnastics can be. It is almost like magic. I can go back and forth between a tensor with its component(s) taken in the coordinate basis or the frame basis. For example, say I have a (2,0) tensor $\tt{A}{}{}$. Then, $ \tt{A}{ \mu m}{}= \tt{A}{nm}{} \tt{E}{\mu}{n}$ , $\tt{A}{m \mu}{}= \tt{A}{ \alpha \mu}{} E_\alpha{}^m$ etc. We can easily show these two, and we hope this is enough to convince a normal person about the general validity of this method. 

\begin{align*}
    \tt{A}{\mu m }{} &= A \lera{E^\mu, dx^m} \\
    &= A \lera{E^\mu{}_n dx^n, dx^m} \\
    &= E^\mu{}_n A \lera{dx^n, dx^m} \\
    &= E^\mu{}_n A^{nm} \\
\end{align*}
and, for the second equality,

\begin{align*}
    \tt{A}{m \mu}{} &= A \lera{dx^m, E^\mu} \\
    &= A \lera{E_\nu{}^m E^\nu, E^\mu} \\
    &= E_\nu{}^m A \lera{ \tt{E}{ \nu}{}, \tt{E}{ \mu}{} } \\
    &= E_\nu{}^m \tt{A}{ \nu \mu}{}
\end{align*}







\subsection{Connection 1-form and Ricci Rotation Coefficient}

Since $ \tt{E}{}{ \mu}$ is a field (at least in a chart U), I can take the covariant derivative of it with respect to the basis vector fields of that chart. The result will have to be another vector field and can be expanded in terms of $ \{ \tt{E}{}{ \mu}\}$.

\begin{align*}
    &\cov{ \tt{E}{}{ \mu}}{ \partial_a} = \tt{ \omega}{ \nu}{ \mu a} \tt{E}{}{ \nu} \numberthis \label{dummy7}
\end{align*}

We can show that although $ \tt{\omega}{\nu}{\mu a}$ is a 1-form in the index a, it is not a tensor in the indices $ \mu$ and $ \nu$. First, notice that the definition of $ \tt{\omega}{\nu}{\mu a}$ can be given as, $ \tt{\omega}{\nu}{\mu a}=\tt{E}{ \nu}{} \lera{ \cov{ \tt{E}{}{ \mu}}{ \partial_a}}$. Now, say I change the chart, but not the frame $ \{ \tt{E}{}{ \mu}\}$. So,

\begin{align*}
    \tt{\omega}{\nu}{\mu a'} =& \tt{E}{ \nu}{} \lera{ \cov{ \tt{E}{}{ \mu}}{ \partial'_a}}\\
    =&\tt{E}{ \nu}{} \lera{ \cov{ \tt{E}{}{ \mu}}{ \lera{ \pp{ x^m}{x'^a} \partial_m}}}\\
    =&\tt{E}{ \nu}{} \lera{ \pp{ x^m}{x'^a}\cov{ \tt{E}{}{ \mu}}{  \partial_m}}\\ 
    =& \pp{ x^m}{x'^a}  \tt{E}{ \nu}{}\lera{ \cov{ \tt{E}{}{ \mu}}{ \partial_m}}\\
    =& \pp{ x^m}{x'^a} \tt{\omega}{\nu}{\mu m}
\end{align*}

This proves that $ \tt{\omega}{\nu}{\mu a}$ is a 1-form in the index a. To show that it is NOT a tensor in the Greek indices, I now change only the frame and not the chart, then,

\begin{align*}
    \tt{\omega}{\nu'}{\mu' a} &= \tt{E'}{ \nu}{} \lera{ \cov{ \tt{E'}{}{ \mu}}{ \partial_a}} \\
    &= \tt{ \Lambda}{ \nu}{ \alpha} \tt{E}{ \alpha}{} \lera{ \cov{ \lera{ \Lambda_ \mu{}^ \beta E_ \beta}}{ \partial_a}} \\
    &= \tt{ \Lambda}{ \nu}{ \alpha} \tt{E}{ \alpha}{} \lera{ \Lambda_ \mu{^ \beta } \cov{\tt{E}{}{\beta}}{a}+ \tt{E}{}{\beta} \cov{\du{\Lambda}{\mu}{\beta}}{a}}\\
     &= \tt{ \Lambda}{ \nu}{ \alpha} \tt{E}{ \alpha}{} \lera{ \Lambda_ \mu{^ \beta } \tt{\omega}{\rho}{\beta a} \tt{E}{}{\rho}+ \tt{E}{}{\beta} \cov{\du{\Lambda}{\mu}{\beta}}{a}}\\
     &= \tt{\Lambda}{ \nu}{\alpha} \du{\Lambda}{\mu}{\beta} \tt{\omega}{\rho}{\beta a} \kron{\alpha}{\rho}+ \tt{\Lambda}{\nu}{\alpha} \partial_a \du{\Lambda}{\mu}{\beta} \kron{\alpha}{\beta}\\
     &= \tt{\Lambda}{ \nu}{\alpha} \du{\Lambda}{\mu}{\beta} \tt{\omega}{\alpha}{\beta a}+ \tt{\Lambda}{\nu}{\alpha} \partial_a \du{\Lambda}{\mu}{\alpha} \\
\end{align*}
The last term shows the inhomogeneous nature of the transformation of $ \tt{\omega}{\nu}{\mu a}$ under LT of the frame.

Our next task is to write down expressions for $ \cov{\tt{A}{\mu}{\nu}}{a}$ just like we know how to expand $ \cov{ A^b{}_c}{a}$ etc. To do that, we follow steps very similar to our treatment of covariant derivatives with Christoffel symbols. In \cref{dummy7} we defined the functions $ \tt{ \omega}{ \mu}{ \nu a}$ as $ \cov{ \tt{E}{}{ \mu}}{ \partial_a} = \tt{ \omega}{ \nu}{ \mu a} \tt{E}{}{ \nu}$. Now we define $ \cov{ \tt{E}{\mu}{}}{ \partial_a} = \tt{ \Omega}{ \mu}{ \nu a} \tt{E}{\nu}{}$. We will show below that, $ \tt{ \Omega}{ \mu}{ \nu a} =- \tt{ \omega}{ \mu}{ \nu a}$. 


\begin{align*}
	& \nabla_{\partial_a}\lera{\tt{E}{\mu}{}\lera{\tt{E}{}{\nu}}} = \lera{\cov{\tt{E}{\mu}{}}{\partial_a}}\lera{\tt{E}{}{\nu}} + \tt{E}{\mu}{}\lera{\cov{\tt{E}{}{\nu}}{\partial_a}} \\
	\Rightarrow\; & \nabla_{\partial_a}  \lera{ \kron{ \mu}{ \nu}}= \tt{\Omega}{\mu}{\rho a} \tt{E}{\rho}{}\lera{\tt{E}{}{\nu}} + \tt{E}{\mu}{}\lera{\tt{\omega}{\rho}{\mu a} \tt{E}{}{\rho}} \\
	\Rightarrow\; & 0 = \tt{\Omega}{\mu}{\rho a} \kron{\rho}{\nu} + \tt{\omega}{\rho}{\mu a} \kron{\mu}{\rho} \\
	\Rightarrow\; & 0 = \tt{\Omega}{\mu}{\nu a}  + \tt{\omega}{\mu}{\nu a} \\
	\Rightarrow\; & \tt{\Omega}{\mu}{\nu a} = - \tt{\omega}{\mu}{\nu a}
\end{align*}

This just means,

\begin{align*}
    \cov{ \tt{E}{\mu}{}}{ \partial_a} = -\tt{ \omega}{ \mu}{ \nu a} \tt{E}{\nu}{} \numberthis \label{dummy8}
\end{align*}



For a (1,1)-tensor $\tt{A}{\mu}{\nu}$, we have $\nabla_a \tt{A}{\mu}{\nu} = \tt{\lera{\nabla_{\partial_a} A}}{\mu}{\nu} $.


\begin{align*}
	\nabla_{\partial_a} A &= \nabla_{\partial_a}\lera{\tt{A}{\mu}{\nu} \tt{E}{}{\mu} \otimes \tt{E}{\nu}{}} \\
	&= \lera{\partial_a \tt{A}{\mu}{\nu}} \tt{E}{}{\mu} \otimes \tt{E}{\nu}{} + \tt{A}{\mu}{\nu} \lera{\cov{\tt{E}{}{\mu}}{\partial_a}} \otimes \tt{E}{\nu}{} + \tt{A}{\mu}{\nu} \tt{E}{}{\mu} \otimes \lera{\cov{\tt{E}{\nu}{}}{\partial_a}} \\
	&= \partial_a \tt{A}{\mu}{\nu} \tt{E}{}{\mu} \otimes \tt{E}{\nu}{} + \tt{A}{\mu}{\nu} \tt{\omega}{\rho}{\mu a} \tt{E}{}{\rho} \otimes \tt{E}{\nu}{} - \tt{A}{\mu}{\nu} \tt{\omega}{\nu}{\rho a} \tt{E}{}{\mu} \otimes \tt{E}{\rho}{} \\
	&= \lera{\partial_a \tt{A}{\mu}{\nu} + \tt{\omega}{\mu}{\rho a} \tt{A}{\rho}{\nu} - \tt{\omega}{\rho}{\nu a} \tt{A}{\mu}{\rho}} \tt{E}{}{\mu} \otimes \tt{E}{\nu}{}
\end{align*}

Therefore,
\begin{align*}
    \nabla_a \tt{A}{\mu}{\nu} = \partial_a \tt{A}{\mu}{\nu} + \tt{\omega}{\mu}{\rho a} \tt{A}{\rho}{\nu} - \tt{\omega}{\rho}{\nu a} \tt{A}{\mu}{\rho} \numberthis \label{dummy9}
\end{align*}




















One immediate inquiry is the expression of $ \tt{\omega}{\nu}{\mu a}$ in terms of $ \cris{a}{bc}$. We know that, 

\begin{align*}
    &\cov{ \tt{A}{ \mu}{}}{ a} =  \lera{\cov{ A^b}{a}} \tt{E}{ \mu}{b}\\
    & \partial_a \lera{ \tt{A}{ \mu}{}} + \tt{\omega}{\mu}{\rho a} \tt{A}{ \rho}{} = \lera{ \partial_a \tt{A}{b}{} + \cris{b}{ma} \tt{A}{m}{}} \tt{E}{ \mu }{b} \\
    & \partial_a \lera{ \tt{E}{ \mu}{m} A^m} + \tt{\omega}{\mu}{\rho a} \tt{A}{ \rho}{} = \lera{ \partial_a \tt{A}{b}{} + \cris{b}{ma} \tt{A}{m}{}} \tt{E}{ \mu }{b} \\
    & \partial_a \lera{ \tt{E}{ \mu}{m} A^m} + \tt{\omega}{\mu}{\rho a} \tt{E}{ \rho}{m}\tt{A}{ m}{} = \lera{ \partial_a \tt{A}{b}{} + \cris{b}{ma} \tt{A}{m}{}} \tt{E}{ \mu }{b} \\
    & \tt{A}{ m}{} \partial_a \tt{E}{ \mu}{m} + \tt{E}{ \mu}{m} \partial_a \tt{A}{m}{} + \tt{\omega}{\mu}{\rho a} \tt{E}{ \rho}{m}\tt{A}{ m}{} = \tt{E}{ \mu}{m } \partial_a \tt{A}{m}{} + \cris{n}{ ma} \tt{E}{ \mu}{n} \tt{A}{m}{}\\
    & \tt{A}{ m}{} \partial_a \tt{E}{ \mu}{m} + \tt{\omega}{\mu}{\rho a} \tt{E}{ \rho}{m}\tt{A}{ m}{} =  \cris{n}{ ma} \tt{E}{ \mu}{n} \tt{A}{m}{}\\
    &  \partial_a \tt{E}{ \mu}{m} + \tt{\omega}{\mu}{\rho a} \tt{E}{ \rho}{m} =  \cris{n}{ ma} \tt{E}{ \mu}{n} \\
    & \text{Now simply multiply by $ \du{E}{ \nu}{m}$ on both sides.}\\
    &\du{E}{ \nu}{m}\partial_a \tt{E}{ \mu}{m} + \tt{\omega}{\mu}{\rho a} \tt{E}{ \rho}{m}\du{E}{ \nu}{m} =  \cris{n}{ ma} \tt{E}{ \mu}{n} \du{E}{ \nu}{m}\\
    &\du{E}{ \nu}{m}\partial_a \tt{E}{ \mu}{m} + \tt{\omega}{\mu}{\rho a} \kron{ \rho}{ \nu} =  \cris{n}{ ma} \tt{E}{ \mu}{n} \du{E}{ \nu}{m}\\
    & \tt{\omega}{\mu}{\nu a} = \cris{n}{ma} \tt{E}{ \mu}{n} \du{E}{ \nu}{m} - \lera{\partial_a \tt{E}{ \mu}{m}} \du{ E}{ \nu}{m} \numberthis \label{omegaintermsofcris}\\
\end{align*}


Similarly, multiplying both sides of \cref{omegaintermsofcris} with $ \du{E}{ \mu}{p} \tt{E}{ \nu}{q}$ so that the two E's with the Christoffel symbol become Kronecker deltas and ultimately go away, 

\begin{align*}
    &\tt{\omega}{\mu}{\nu a} \du{E}{ \mu}{p} \tt{E}{ \nu}{q} = \cris{n}{ma} \du{E}{ \mu}{p } \tt{E}{ \mu}{n} \du{E}{ \nu}{m} \tt{E}{ \nu}{q} - \partial_a \tt{E}{ \mu}{m} \; \du{E}{ \nu}{m} \tt{E}{ \nu}{q} \du{E}{ \mu}{p} \\
    & \tt{\omega}{\mu}{\nu a} \du{E}{ \mu}{p} \tt{E}{ \nu}{q} = \cris{n}{ma} \kron{p}{n} \kron{m}{q} - \partial_a \tt{E}{ \mu}{m} \kron{m}{q} \du{E}{ \mu}{p} \\
    & \tt{\omega}{\mu}{\nu a} \du{E}{ \mu}{p} \tt{E}{ \nu}{q}= \cris{p}{qa} - \partial_a \tt{E}{ \mu}{q} \; \du{E}{ \mu}{p} \\
    & \cris{p}{qa} = \tt{\omega}{\mu}{\nu a} \du{E}{ \mu}{p} \tt{E}{ \nu}{q} + \partial_a \tt{E}{ \mu}{q} \; \du{E}{ \mu}{p} \\
    & \text{Just replace p with m, q with n, and swap $ \mu$ and $ \nu$} \\
    & \cris{m}{na} = \tt{\omega}{\nu}{\mu a} \du{E}{ \nu}{ m} \tt{E}{ \mu}{n} + \lera{\partial_a \tt{E}{ \nu}{n} } \du{E}{ \nu}{m} \numberthis \label{crisintermsofomega} \\
\end{align*}


This leaves one more quantity to define in this section, called the Ricci rotation coefficients. It is basically the connection 1-form expressed in the tetrad basis instead of the coordinate basis. That is, while a connection 1-form $ \tt{\omega}{\nu}{\mu a}$ has the 1-form index ``a'' from the coordinate basis, the Ricci rotation coefficient $ \tt{\omega}{\nu}{\mu \rho}$ is the connection 1-form expanded in terms of $ \{ \tt{E}{ \mu}{}\}$. 

\begin{align*}
    &\tt{\omega}{\nu}{\mu \rho} = \tt{\omega}{\nu}{\mu a} \du{E}{ \rho}{a} \numberthis \label{riccirotationdefined}
\end{align*}

Another way to express the Ricci rotation coefficient is, $ \tt{\omega}{\nu}{\mu \rho} = \tt{E}{ \nu}{} \lera{ \cov{ \tt{E}{}{ \mu}}{ \tt{E}{}{ \rho}}}$. We can quickly see that as follows, by starting with the RHS,

\begin{align*}
    \tt{E}{ \nu}{} \lera{ \cov{ \tt{E}{}{ \mu}}{ \tt{E}{}{ \rho}}} &= \tt{E}{ \nu}{} \lera{ \cov{ \tt{E}{}{ \mu}}{ \lera{ \du{E}{ \rho}{m} \partial_m}}} \\
    &= \tt{E}{ \nu}{} \lera{ \du{E}{ \rho}{m} \cov{ \tt{E}{}{ \mu}}{ \partial_m}} \\
    &=  \du{E}{ \rho}{m}\tt{E}{ \nu}{} \lera{ \cov{ \tt{E}{}{ \mu}}{ \partial_m}} \\
    &= \du{E}{ \rho}{m} \tt{\omega}{\nu}{\mu m} \\
    &= \tt{\omega}{\nu}{\mu \rho} \\
\end{align*}


Now, our task is to write an expression for $ \cov{\tt{A}{\mu}{\nu}}{ \alpha}$ just like we showed how to expand $ \cov{ A^\mu{}_\nu}{a}$ etc. To do that, we follow steps very similar to what we have done in the previous case. First, we claim that $\nabla_{\tt{E}{}{\alpha}} \tt{E}{\beta}{} = - \tt{\omega}{\beta}{\mu\alpha} \tt{E}{\mu}{}$. This is easily shown, $ \nabla_{\tt{E}{}{\alpha}} \tt{E}{\beta}{}= \nabla_{ \lera{E_\alpha{}^a \partial_a}} \tt{E}{\beta}{} = E_\alpha{}^a \nabla_{\partial_a}E^{\beta}= -E_\alpha{}^a \omega^{ \beta}{}_{ \mu a} E^{\mu}= - \tt{ \omega}{ \beta}{ \mu \alpha} E^{\mu} $.







Take the $(1,1)$ tensor $\tt{A}{\mu}{\nu}$. We know that, $\nabla_\alpha \tt{A}{\mu}{\nu} = \tt{\lera{\nabla_{\tt{E}{}{\alpha}} A}}{\mu}{\nu} $. Then,
\begin{align*}
	\nabla_\alpha A &= \nabla_{\tt{E}{}{\alpha}} A \\
	&= \nabla_{\tt{E}{}{\alpha}}\lera{\tt{A}{\mu}{\nu} \tt{E}{}{\mu} \otimes \tt{E}{\nu}{}} \\
	&= \tt{E}{}{\alpha}\lera{\tt{A}{\mu}{\nu}} \tt{E}{}{\mu} \otimes \tt{E}{\nu}{} + \tt{A}{\mu}{\nu} \lera{\nabla_{\tt{E}{}{\alpha}} \tt{E}{}{\mu}} \otimes \tt{E}{\nu}{} + \tt{A}{\mu}{\nu} \tt{E}{}{\mu} \otimes \nabla_{\tt{E}{}{\alpha}} \tt{E}{\nu}{} \\
	&= \du{E}{\alpha}{a} \partial_a \tt{A}{\mu}{\nu} \tt{E}{}{\mu} \otimes \tt{E}{\nu}{} + \tt{A}{\mu}{\nu} \tt{\omega}{\rho}{\mu\alpha} \tt{E}{}{\rho} \otimes \tt{E}{\nu}{}- \tt{A}{\mu}{\nu} \tt{\omega}{\nu}{\rho\alpha} \tt{E}{}{\mu} \otimes \tt{E}{\rho}{} \\
	&= \lera{\du{E}{\alpha}{a} \partial_a \tt{A}{\mu}{\nu} + \tt{\omega}{\mu}{\rho\alpha} \tt{A}{\rho}{\nu} - \tt{\omega}{\rho}{\nu\alpha} \tt{A}{\mu}{\rho}} \tt{E}{}{\mu} \otimes \tt{E}{\nu}{}
\end{align*}
Hence,
\begin{align*}
	\nabla_\alpha \tt{A}{\mu}{\nu} = \partial_\alpha \tt{A}{\mu}{\nu} + \tt{\omega}{\mu}{\rho\alpha} \tt{A}{\rho}{\nu} - \tt{\omega}{\rho}{\nu\alpha} \tt{A}{\mu}{\rho} \numberthis \label{dummy10}
\end{align*}
where I defined, $\partial_\alpha \tt{A}{\mu}{\nu} \equiv \du{E}{\alpha}{a} \partial_a \tt{A}{\mu}{\nu}$. We can always go back and forth between different kinds of indices. For example, for a $(0,1)$-tensor $A$,
\begin{align*}
	\nabla_\mu A_\nu &= \nabla_a A_b \du{E}{\mu}{a} \du{E}{\nu}{b} \\
	\nabla_a A_b &= \nabla_\mu A_\nu \tt{E}{\mu}{a} \tt{E}{\nu}{b}
\end{align*}

We can define the fully lowered version of the Ricci rotation coefficients by lowering the first index using $ \eta_{ \mu \nu}$. This $ \tt{ \omega }{ }{ \mu \nu \alpha}$ is captured by the following alternate definition. 

\begin{align*}
    \omega_{\alpha\beta\mu} := g\left(  \nabla_\mu E_\beta , E_\alpha\right) \numberthis \label{dummy14}
\end{align*}

Now we need to show that this alternate definition captures the definition of $ \tt{ \omega }{ }{ \alpha \beta \mu }= \eta_{ \alpha \rho} \tt{ \omega}{ \rho}{ \beta \mu}$.

\begin{align*}
\omega_{ \alpha \beta \mu } &= g\left( E_{ \alpha }, \nabla_{ \mu} E_{ \beta}\right) \\
&= g_{ab } E_{\alpha}{}^{a}\, \omega^{\rho}{}_{\beta \mu} E_{\rho}{}^{b} \\
&= E_{\alpha b} E_{\rho}{}^{b}\, \omega^{\rho}{}_{\beta \mu} \\
&= \eta_{\alpha \sigma} E^{\sigma}{}_{b} E_{\rho}{}^{b}\, \omega^{\rho}{}_{ \beta \mu} \\
&= \eta_{\alpha \sigma}\, \delta^{\sigma}_{\rho}\, \omega^{\rho}{}_{\beta \mu} \\
&= \eta_{\alpha \rho}\, \omega^{\rho}{}_{\beta  \mu} \\
&= \omega_{\alpha \beta \mu}
\end{align*}

An important property of the Ricci rotation coefficients is the antisymmetry in the first two indices. $ \tt{ \omega}{}{ \mu \nu \alpha}= -\tt{ \omega}{}{ \nu \mu \alpha}$


\begin{proof}
    Since the covariant derivative is compatible with the metric $ g_{ab}$, we have $ \nabla_a g_{\mu\nu} = \lera{\nabla_a g}_{\mu\nu} = \lera{\nabla_{\partial_a} g}_{\mu\nu} = 0$.
    \begin{align*}
        & \nabla_a \tt{g}{}{ \mu \nu} =0 \\
	\Rightarrow\; & \partial_a \eta_{\mu\nu} - \tt{\omega}{\rho}{\mu a} \eta_{\rho\nu} - \tt{\omega}{\rho}{\nu a} \eta_{\mu\rho} = 0 \\
	\Rightarrow\; & 0 - \omega_{\nu\mu a} = \omega_{\mu\nu a} \\
	\Rightarrow\; & \omega_{\mu\nu a} = - \omega_{\nu\mu a} \\
	\Rightarrow\; & \omega_{\mu\nu\alpha} = - \omega_{\nu\mu\alpha}
\end{align*}


\end{proof}

The crucial step in the proof was the constancy of $ \tt{ \eta}{}{ \mu \nu}$ which led to $ \partial_a \tt{ \eta}{}{ \mu \nu}=0$. Any tetrad frame which shows this property is called a \textbf{rigid} tetrad. 
	

Based on the antisymmetry property, there are 24 independent Ricci rotation coefficients. 

The following proposition will help us in some of the calculations. It expresses the covariant derivative of tetrad vectors in terms of the Ricci rotation coefficients.




\begin{proposition}
$\nabla_a E_{\mu k} = E^{\beta}{}_a\, \omega_{\alpha\mu\beta} E^{\alpha}{}_k $
\end{proposition}

\begin{proof}\begin{align*}
\nabla_a E_{\mu k} &= \left( \nabla_a E_\mu \right)^m g_{mk} \\
&= \left( \omega^{\alpha}{}_{\mu a} E_\alpha \right)^m g_{mk} \\
&= \omega^{\alpha}{}_{\mu a} E_\alpha{}^m g_{mk} \\
&= \omega^{\alpha}{}_{\mu a} E_{\alpha k} \\
&= \omega_{\alpha\mu a} E^{\alpha}{}_k \\
&= E^{\beta}{}_a\, \omega_{\alpha\mu\beta} E^{\alpha}{}_k
\end{align*}

\end{proof}

\subsubsection{Lie Bracket of Tetrads and Structure Constants}


Since the tetrads furnish a set of four vector fields, we can take the Lie bracket of these fields and see what we get.
\[
\left[ E_\alpha, E_\beta \right] = C^{\mu}{}_{\alpha\beta} E_\mu
\]
The coefficients $C^{\mu}{}_{\alpha\beta}$ are called structure constants. These constants can be expressed in terms of the Ricci rotation coefficients. To see how, consider the Lie bracket acting on a function $f$.
\begin{align*}
\left[ E_\alpha, E_\beta \right] f &= E_\alpha\left( E_\beta(f) \right) - E_\beta\left( E_\alpha(f) \right) \\
&= E_\alpha\left( E_\beta{}^m \partial_m f \right) - E_\beta\left( E_\alpha{}^m \partial_m f \right) \\
&= E_\alpha\left( E_\beta{}^m \right) \partial_m f + E_\beta{}^m E_\alpha\left( \partial_m f \right) - E_\beta\left( E_\alpha{}^m \right) \partial_m f - E_\alpha{}^m E_\beta\left( \partial_m f \right) \\
&= E_\alpha{}^n \partial_n E_\beta{}^m \partial_m f + E_\beta{}^m E_\alpha{}^n \partial_n \partial_m f - E_\beta{}^n \partial_n E_\alpha{}^m \partial_m f - E_\alpha{}^m E_\beta{}^n \partial_n \partial_m f \\
&= \left( E_\alpha{}^n \partial_n E_\beta{}^m - E_\beta{}^n \partial_n E_\alpha{}^m \right) \partial_m f + E_\beta{}^m E_\alpha{}^n \partial_n \partial_m f - E_\beta{}^m E_\alpha{}^n \partial_m \partial_n f \\
&= \left( E_\alpha{}^n \partial_n E_\beta{}^m - E_\beta{}^n \partial_n E_\alpha{}^m \right) \partial_m f \\
& \text{We can replace the ordinary derivatives with covariant derivatives.}\\
&= \left( E_\alpha{}^n \nabla_n E_\beta{}^m - E_\beta{}^n \nabla_n E_\alpha{}^m \right) \partial_m f \\
&= \left( E_\alpha{}^n \left( \nabla_n E_\beta \right)^m - E_\beta{}^n \left( \nabla_n E_\alpha \right)^m \right) \partial_m f \\
&= \left( E_\alpha{}^n\, \omega^{\mu}{}_{\beta n} E_\mu{}^m - E_\beta{}^n\, \omega^{\mu}{}_{\alpha n} E_\mu{}^m \right) \partial_m f \\
&= \left( \omega^{\mu}{}_{\beta\alpha} - \omega^{\mu}{}_{\alpha\beta} \right) E_\mu{}^m \partial_m f \\
&= \left( \omega^{\mu}{}_{\beta\alpha} - \omega^{\mu}{}_{\alpha\beta} \right) E_\mu(f) \\
\Rightarrow \left[ E_\alpha, E_\beta \right] f &= \left( \omega^{\mu}{}_{\beta\alpha} - \omega^{\mu}{}_{\alpha\beta} \right) E_\mu(f) \\
\Rightarrow C^{\mu}{}_{\alpha\beta} E_\mu(f) &= \left( \omega^{\mu}{}_{\beta\alpha} - \omega^{\mu}{}_{\alpha\beta} \right) E_\mu(f) \\
\Rightarrow C^{\mu}{}_{\alpha\beta} &= \omega^{\mu}{}_{\beta\alpha} - \omega^{\mu}{}_{\alpha\beta}
\end{align*}






\subsubsection{The Lambda Symbol}

The $ \Lambda$-symbol is defined as follows: 

\begin{align*}
\Lambda_{\mu\nu\alpha} &:= \omega_{\mu\nu\alpha} - \omega_{\alpha\nu\mu} \\
&= \eta_{\mu\rho}\, \omega^{\rho}{}_{\nu\alpha} - \eta_{\alpha\rho}\, \omega^{\rho}{}_{\nu\mu} \\
&= \eta_{\mu\rho}\, \omega^{\rho}{}_{\nu a} E_\alpha{}^a - \eta_{\alpha\rho}\, \omega^{\rho}{}_{\nu a} E_\mu{}^a \\
&= \eta_{\mu\rho}\, E^{\rho b} \nabla_a E_{\nu b} E_\alpha{}^a - \eta_{\alpha\rho}\, E^{\rho b} \nabla_a E_{\nu b} E_\mu{}^a \\
&= E_\mu{}^b E_\alpha{}^a \nabla_a E_{\nu b} - E_\alpha{}^b E_\mu{}^a \nabla_a E_{\nu b} \\
&= E_\mu{}^a E_\alpha{}^b \nabla_b E_{\nu a} - E_\mu{}^a E_\alpha{}^b \nabla_a E_{\nu b}\\
&= E_\mu{}^a E_\alpha{}^b \left( \nabla_b E_{\nu a} - \nabla_a E_{\nu b} \right)
\end{align*}

Interestingly, if we look at the term in the parentheses, $\left(\nabla_b E_{\nu a} - \nabla_a E_{\nu b}\right)$, it is antisymmetric in $(a,b)$. So, we can actually replace the covariant derivative by the ordinary derivative.
\begin{align*}
\Lambda_{\mu\nu\alpha} := E_\mu{}^a E_\alpha{}^b \left( \partial_b E_{\nu a} - \partial_a E_{\nu b} \right) \numberthis\label{dummy13}
\end{align*}
 
So, we can evaluate the $\Lambda$'s with only ordinary derivatives of the tetrad vectors. On the other hand, we can express $\omega_{\mu\nu\alpha}$ in terms of the $\Lambda$'s.
\begin{align*}
&\text{(a)}  \Lambda_{\mu\nu\alpha} = \omega_{\mu\nu\alpha} - \omega_{\alpha\nu\mu} = \omega_{\mu\nu\alpha} + \omega_{\nu\alpha\mu} \\
&\text{(b)} \Lambda_{\nu\alpha\mu} = \omega_{\nu\alpha\mu} - \omega_{\mu\alpha\nu} = \omega_{\nu\alpha\mu} + \omega_{\alpha\mu\nu} \\
&\text{(c)}  \Lambda_{\alpha\mu\nu} = \omega_{\alpha\mu\nu} - \omega_{\nu\mu\alpha} = \omega_{\alpha\mu\nu} + \omega_{\mu\nu\alpha}
\end{align*}
\begin{align*}
\text{(a)} + \text{(c)} - \text{(b)} \Rightarrow 2\omega_{\mu\nu\alpha} &= \Lambda_{\mu\nu\alpha} + \Lambda_{\alpha\mu\nu} - \Lambda_{\nu\alpha\mu} \\
\Rightarrow \omega_{\mu\nu\alpha} &= \frac{1}{2}\left( \Lambda_{\mu\nu\alpha} - \Lambda_{\nu\alpha\mu}+ \Lambda_{\alpha\mu\nu}  \right) \numberthis \label{dummy15}
\end{align*}

The $ \Lambda$'s are antisymmetric in the first and the last indices. $ \Lambda_{\alpha\nu\mu} = \omega_{\alpha\nu\mu} - \omega_{\mu\nu\alpha} = -\left( \omega_{\mu\nu\alpha} - \omega_{\alpha\nu\mu} \right) = -\Lambda_{\mu\nu\alpha}$.


\subsection{Ricci Identities}

Here our goal is to expand the Riemann tensor in terms of only Ricci rotation coefficients and their derivatives. Start with the definition of the Riemann tensor.  
\[
R(\omega, Z, X, Y) = \omega\left( \nabla_X \nabla_Y Z - \nabla_Y \nabla_X Z - \nabla_{[X,Y]} Z \right)
\]
The tetrad components of the Riemann tensor are given by,
\begin{align*}
R^{\alpha}{}_{\beta\gamma\delta} &= R\left( E^\alpha, E_\beta, E_\gamma, E_\delta \right) \\
&= E^\alpha\left( \nabla_\gamma \nabla_\delta E_\beta - \nabla_\delta \nabla_\gamma E_\beta - \nabla_{[E_\gamma, E_\delta]} E_\beta \right) \\
&= E^\alpha\left( \nabla_\gamma\left( \omega^{\rho}{}_{\beta\delta} E_\rho \right) - \nabla_\delta\left( \omega^{\rho}{}_{\beta\gamma} E_\rho \right) - C^{\mu}{}_{\gamma\delta} \nabla_\mu E_\beta \right)  \\
&= E^\alpha\Big( \left( \partial_\gamma \omega^{\rho}{}_{\beta\delta} \right) E_\rho + \omega^{\rho}{}_{\beta\delta} \nabla_\gamma E_\rho - \left( \partial_\delta \omega^{\rho}{}_{\beta\gamma} \right) E_\rho  - \omega^{\rho}{}_{\beta\gamma} \nabla_\delta E_\rho - C^{\mu}{}_{\gamma\delta}\, \omega^{\sigma}{}_{\beta\mu} E_\sigma \Big) \\
&= \partial_\gamma \omega^{\rho}{}_{\beta\delta}\, \delta^{\alpha}_{\rho} - \partial_\delta \omega^{\rho}{}_{\beta\gamma}\, \delta^{\alpha}_{\rho} + \omega^{\rho}{}_{\beta\delta}\, \omega^{\mu}{}_{\rho\gamma}\, \delta^{\alpha}_{\mu}  - \omega^{\rho}{}_{\beta\gamma}\, \omega^{\mu}{}_{\rho\delta}\, \delta^{\alpha}_{\mu} - C^{\mu}{}_{\gamma\delta}\, \omega^{\sigma}{}_{\beta\mu}\, \delta^{\alpha}_{\sigma} \\
&= \partial_\gamma \omega^{\alpha}{}_{\beta\delta} - \partial_\delta \omega^{\alpha}{}_{\beta\gamma} + \omega^{\alpha}{}_{\rho\gamma}\, \omega^{\rho}{}_{\beta\delta} - \omega^{\alpha}{}_{\rho\delta}\, \omega^{\rho}{}_{\beta\gamma} - C^{\rho}{}_{\gamma\delta}\, \omega^{\alpha}{}_{\beta\rho} \\
&= \partial_\gamma \omega^{\alpha}{}_{\beta\delta} - \partial_\delta \omega^{\alpha}{}_{\beta\gamma} + \omega^{\alpha}{}_{\rho\gamma}\, \omega^{\rho}{}_{\beta\delta} - \omega^{\alpha}{}_{\rho\delta}\, \omega^{\rho}{}_{\beta\gamma} - \left( \omega^{\rho}{}_{\delta\gamma} - \omega^{\rho}{}_{\gamma\delta} \right) \omega^{\alpha}{}_{\beta\rho} \\
&= \partial_\gamma \omega^{\alpha}{}_{\beta\delta} - \partial_\delta \omega^{\alpha}{}_{\beta\gamma} + \omega^{\alpha}{}_{\rho\gamma}\, \omega^{\rho}{}_{\beta\delta} - \omega^{\alpha}{}_{\rho\delta}\, \omega^{\rho}{}_{\beta\gamma} + \omega^{\alpha}{}_{\beta\rho}\, \omega^{\rho}{}_{\gamma\delta} - \omega^{\alpha}{}_{\beta\rho}\, \omega^{\rho}{}_{\delta\gamma} \\
\Rightarrow \tt{R}{}{ \alpha \beta \gamma \delta } &= \partial_{ \gamma} \tt{ \omega}{}{ \alpha \beta \delta} - \partial_{ \delta} \tt{ \omega}{}{ \alpha \beta \gamma} + \tt{ \omega}{}{ \alpha \rho \gamma}\, \tt{ \omega}{ \rho}{ \beta \delta} - \tt{ \omega}{}{ \alpha \rho \delta}\, \tt{ \omega}{ \rho}{ \beta \gamma} + \tt{ \omega}{}{ \alpha \beta \rho}\, \tt{ \omega}{ \rho}{ \gamma \delta} - \tt{ \omega}{}{ \alpha \beta \rho}\, \tt{ \omega}{ \rho}{ \delta \gamma}\\
\Rightarrow \tt{R}{}{ \alpha \beta \gamma \delta } &= \partial_{ \gamma} \tt{ \omega}{}{ \alpha \beta \delta} - \partial_{ \delta} \tt{ \omega}{}{ \alpha \beta \gamma} + \tt{ \omega}{}{  \rho \alpha \delta}\, \tt{ \omega}{ \rho}{ \beta \gamma} -   \tt{ \omega}{}{  \rho \alpha \gamma}\, \tt{ \omega}{ \rho}{ \beta \delta} + \tt{ \omega}{}{ \alpha \beta \rho}\, \tt{ \omega}{ \rho}{ \gamma \delta} - \tt{ \omega}{}{ \alpha \beta \rho}\, \tt{ \omega}{ \rho}{ \delta \gamma} \numberthis \label{dummy16}
\end{align*}

In the penultimate line the lowering of the $ \alpha$ index is allowed because of the tetrad is rigid, and so, for example, $ \tt{ \eta }{}{ \alpha \sigma} \partial_{ \gamma} \tt{ \omega}{ \sigma}{ \beta \delta}= \partial_{ \gamma} \lera{ \tt{ \eta}{}{ \alpha \sigma} \tt{ \omega}{ \sigma}{ \beta \delta}}= \partial_{ \gamma} \tt{ \omega}{}{ \alpha \beta \delta}$.

\subsubsection{Torsion and Curvature 2-forms}
We will first simply define the Torsion and Curvature 2-forms, then show that these definitions work. There are other ways of arriving at these definitions that might be more direct, but since we have to memorize these anyway, we will simply state the definition without motivation. 

The torsion 2-form is defined as (suppressing the Latin indices),

\begin{align*}
    \tt{T}{ \mu }{} = d \tt{E}{ \mu}{} + \tt{\omega}{\mu}{\nu} \wedge \tt{E}{ \nu}{}  \numberthis \label{torsionform}\\
\end{align*}

The curvature 2-form is defined as (again, suppressing the Latin indices),

\begin{align*}
    \tt{R}{ \mu}{ \nu} = d \tt{\omega}{\mu}{\nu} + \tt{\omega}{\mu}{\alpha} \wedge \tt{\omega}{\alpha}{\nu}  \numberthis \label{curvatureform}\\
\end{align*}



To show that the definition of torsion form works, start with expanding the RHS of \cref{torsionform},

\begin{align*}
    d \tt{E}{ \mu}{ab} + \bigl( \tt{\omega}{\mu}{\nu} \wedge \tt{E}{ \nu } {} \bigr)_{ab} &= \partial_a \tt{E}{ \mu}{b} - \partial_b \tt{E}{ \mu}{a} + \tt{\omega}{\mu}{\nu a} \tt{E}{ \nu}{b}-\tt{\omega}{\mu}{\nu b} \tt{E}{ \nu}{a} \numberthis \label{torsionform_rhs_expanded} \\
\end{align*}


Now, \begin{align*}
    \tt{T }{ \mu}{ ab} &= T \lera{ \tt{E}{ \mu}{}, \partial_a, \partial_b} \\
    &= \tt{E}{ \mu}{} \lera{ \cov{ \partial_b}{a}- \cov{ \partial_a}{b}} \\
    &= \tt{E}{ \mu}{} \left(\cov{ \partial_b}{a} \right) - \lera{ a \leftrightarrow b}\\
    &= \tt{E}{ \mu}{} \lera{ \cris{ m}{ba} \partial_m} -\lera{ a \leftrightarrow b} \\
    &= \tt{E}{ \mu}{ } \lera{ \cris{m}{ba} \tt{E}{ \nu}{m} \tt{E}{}{ \nu}} - \lera{ a \leftrightarrow b}\\
    &=\cris{m}{ba} \tt{E}{ \nu}{m} \kron{ \mu}{ \nu} - \lera{ a \leftrightarrow b}\\
    &=\cris{m}{ba} \tt{E}{ \mu}{m} - \lera{ a \leftrightarrow b}\\
    &= \tt{E}{ \mu}{m } \lera{ \tt{\omega}{\beta}{\alpha a} \du{E}{ \beta}{ m} \tt{E}{ \alpha}{b} + \partial_a \tt{E}{ \beta}{b} \; \du{E}{ \beta}{m}} - \lera{ a \leftrightarrow b}\\ 
    &= \tt{\omega}{\beta}{\alpha a} \tt{E}{ \mu}{m} \du{E}{ \beta}{m } \tt{E}{ \alpha}{b} + \partial_a \tt{E}{ \beta}{b} \;\tt{E}{ \mu}{m} \du{E}{ \beta}{m} - \lera{ a \leftrightarrow b}\\
    &= \tt{\omega}{\beta}{\alpha a} \kron{ \mu}{ \beta}\tt{E}{ \alpha}{b} + \partial_a \tt{E}{ \beta}{b} \; \kron{ \mu}{ \beta} - \lera{ a \leftrightarrow b}\\
    &= \partial_a \tt{E}{ \mu}{b} +\tt{\omega}{\mu}{\alpha a} \tt{E}{ \alpha}{b} - \lera{ a \leftrightarrow b}\\
    &= \partial_a \tt{E}{ \mu}{b} +\tt{\omega}{\mu}{\nu a} \tt{E}{ \nu}{b} -\partial_b \tt{E}{ \mu}{a} -\tt{\omega}{\mu}{\nu b} \tt{E}{ \nu}{a} \\
    &=\partial_a \tt{E}{ \mu}{b}-\partial_b \tt{E}{ \mu}{a}+\tt{\omega}{\mu}{\nu a} \tt{E}{ \nu}{b}-\tt{\omega}{\mu}{\nu b} \tt{E}{ \nu}{a}\\
    & \text{This matches \cref{torsionform_rhs_expanded}}
\end{align*}
