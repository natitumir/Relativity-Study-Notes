\section{Topological Space}

\begin{definition}
    Let M be a set. We define a topology $ \mathcal{O}$ on M to be a collection of subsets of M such that
    \begin{enumerate}[label=(\roman*)]
        \item $ \emptyset, M \in \mathcal{O}$
        \item $ \mathcal{O}$ is closed under arbitrary union and finite intersection. i.e.
    \end{enumerate}

    \begin{align*}
        & U, V \in \mathcal{O} \implies U \cap V \in \mathcal{O} \\
        & U_\alpha \in \mathcal{O} \lera{ \alpha \in A} \implies \bigcup_{\alpha \in A} U_\alpha \in \mathcal{O} \\
    \end{align*}
\end{definition}

\begin{definition}
    A set M equipped with a topology $ \mathcal{O}$, i.e. $ \lera{ M, \mathcal{O}}$ is called a topological space.
\end{definition}

Examples of topological spaces are as follows. For any set M, the collection $ \mathcal{O} = \{ \emptyset, M \}$ is a topology. This is called the chaotic topology. The collection $ \mathcal{O} = P(M)$ (Power set of M) is called the discrete topology. Another example is $ \lera{M, \mathcal{O}}$ with $M = \{ 1,2,3 \} $ and $ \mathcal{O} = \{ \{1\}, \{2\}, \{1,2\}, \{1,2,3\}, \emptyset \} $



Consider $ \mathbb{R}^n$ where we define the ``open ball/soft ball'' of radius r around a point $ p \in \mathbb{R}^n$ as:

\begin{align*}
    & B_r(p) := \left\{ x \in \mathbb{R}^n \; \middle| \; \sum_{i=1}^{n} | x^i - p^i |^2 < r^2 \right\} \\
\end{align*}

With the help of this we can define a topology $ \mathcal{O}_{std}$ on $ \mathbb{R}^n$.

\begin{definition}
    A set $ U \in \mathcal{O}_{std}$ iff $ U \subseteq \mathbb{R}^n$ and

    \begin{align*}
        & p \in U \implies \exists r \in \mathbb{R}^+ : B_r(p) \subseteq U \\
    \end{align*}
\end{definition}

\begin{proposition}
    $ \lera{ \mathbb{R}^n, \mathcal{O}_{std}}$ is a topological space.
\end{proposition}

\begin{proof}
    $ \emptyset \in \mathcal{O}_{std}$ since $ p \in \emptyset \implies \exists r \in \mathbb{R}^+ : B_r(p) \subseteq \emptyset$ (vacuously)

    $ \mathbb{R}^n \in \mathcal{O}_{std}$ since $ p \in \mathbb{R}^n \implies \exists r = 1 \in \mathbb{R}^+ : B_r(p) \subseteq \mathbb{R}^n$

    If $ U, V \in \mathcal{O}_{std}$ then assume $ p \in U \cap V$

    \begin{align*}
        p \in U \cap V &\implies p \in U \text{ and } p \in V \\
        &\implies \exists r_1, r_2 \in \mathbb{R}^+ : B_{r_1}(p) \subseteq U \text{ and } B_{r_2}(p) \subseteq V \\
    \end{align*}

    Now, $ r := \min \lera{ r_1, r_2}$

    \begin{align*}
        & \therefore B_r(p) \subseteq U \text{ and } B_r(p) \subseteq V \\
        & \implies B_r(p) \subseteq U \cap V \\
        & \therefore U, V \in \mathcal{O}_{std} \implies \lera{ U \cap V} \in \mathcal{O}_{std} \\
    \end{align*}

    Assume $ U_\alpha \in \mathcal{O}_{std}$ $ \lera{ \alpha \in A}$

    Let $ p \in \bigcup_{\alpha \in A} U_\alpha$

    \begin{align*}
        & \implies p \in U_\beta \text{ for some } \beta \in A \\
        & \implies \exists r \in \mathbb{R}^+ : B_r(p) \subseteq U_\beta \\
        & \implies \exists r \in \mathbb{R}^+ : B_r(p) \subseteq \bigcup_{\alpha \in A} U_\alpha \\
        & \therefore U_\alpha \in \mathcal{O}_{std} \lera{ \alpha \in A} \implies \lera{ \bigcup_{\alpha \in A} U_\alpha} \in \mathcal{O}_{std} \\
    \end{align*}
\end{proof}

So, in short, we can say that $ \lera{ \mathbb{R}^n, \mathcal{O}_{std}}$ is a topological space.

\subsection*{Basis}

The basic idea is to have a collection of subsets from which we can take unions and construct any set $ U \in \mathcal{O}$.

\begin{definition}
    A collection $ \mathcal{B}$ of open sets is a basis for the topology $ \mathcal{O}$ if given any set $ U \in \mathcal{O}$ and a point $ p \in U$, there is an open set $ B \in \mathcal{B}$ such that $ p \in B \subseteq U$.
\end{definition}

An element of the basis $ \mathcal{B}$ is called a basic open set.

\begin{theorem}
    A collection of open sets $ \mathcal{B}$ is a basis of $ \mathcal{O}$ iff any open set U is a union of sets of $ \mathcal{B}$.
\end{theorem}

\begin{proof}
    $ \lera{ \implies}$ We assume that $ \mathcal{B}$ is a basis for $ \mathcal{O}$. So, if $ U \in \mathcal{O}$, then for $ p \in U$, $ \exists B_p \in \mathcal{B}$ such that $ p \in B_p \subseteq U$.

    \begin{align*}
        & \therefore U = \bigcup_{p \in U} B_p \\
    \end{align*}

    $ \lera{ \impliedby}$ Since any open set U is a union of sets of $ \mathcal{B}$, for any point $ p \in U$, we have a set $ B \in \mathcal{B}$ such that $ p \in B \subseteq U$.
\end{proof}

$ \mathcal{B}_{rat}$ is the set of all open balls $ B_r(p)$ where $ r \in \mathbb{Q}^+$ and $ p \in \mathbb{Q}^n$, i.e. rational centers and rational radius.

\begin{theorem}
    The set $ \mathcal{B}_{rat}$ is a basis for $ \mathcal{O}_{std}$.
\end{theorem}

\begin{proof}
    Take $ U \in \mathcal{O}_{std}$

    Now, for any point $ p \in U$, there exists $ r \in \mathbb{Q}^+$ such that $ B_r(p) \subseteq U$.

    Now, take a rational point $ q \in B_{r/2}(p)$

    \begin{center}
    \begin{tikzpicture}[scale=1.0]
        \draw (0,0) circle (1.5);
        \draw[dashed] (0,0) circle (0.75);
        \fill (0,0) circle (1.5pt) node[above left] {$ p$};
        \fill (0.35,-0.3) circle (1.5pt) node[above right] {$ q$};
    \end{tikzpicture}
    \end{center}

    So,

    \begin{align*}
        & d(p,q) < \frac{r}{2} \\
    \end{align*}

    Consider the ball $ B_{r/2}(q)$

    \begin{align*}
        & p \in B_{r/2}(q) \numberthis \label{p_in_half_ball} \\
    \end{align*}

    Let, $ x \in B_{r/2}(q)$

    \begin{align*}
        & \implies d(x,q) < \frac{r}{2} \\
    \end{align*}

    Now,

    \begin{align*}
        & d(x,p) \leq d(x,q) + d(q,p) < \frac{r}{2} + \frac{r}{2} = r \\
        & \implies d(x,p) < r \implies x \in B_r(p) \\
        & \therefore p \in B_{r/2}(q) \subseteq B_r(p) \subseteq U \\
    \end{align*}
\end{proof}


So, we can say that $ \mathcal{B}_{rat}$ is a basis for $ \mathcal{O}_{std}$.

There is a curious thing to note here. The set $ \mathcal{B}_{rat}$ of all the rational balls (balls with rational center and rational radii) is a countable set. Since each ball can be indexed by $ \mathbb{Q}^n$ (a countable set) and $ \mathbb{Q}^+$ (another countable set), we have a countable infinity of rational balls.

\begin{definition}
    A topological space $ \lera{ M, \mathcal{O}}$ is called second countable iff $ \mathcal{O}$ has a countable basis.
\end{definition}

\section{Hausdorff Topological Space}

\begin{definition}
    $ \lera{ T, \mathcal{O}_T}$ is called a Hausdorff top. space :=

    \begin{align*}
        \text{iff } \forall p, q \in T \text{ and } p \neq q & \\
        \implies \exists U_p \in \mathcal{O}_T \text{ and } \exists U_q \in \mathcal{O}_T, \; p \in U_p \; \& \; q \in U_q & \\
        \text{and } U_p \cap U_q = \emptyset & \\
    \end{align*}
\end{definition}

In short, for every pair of points, we have disjoint open sets that contain those points.

\begin{definition}
    Let, $ \lera{ M, \mathcal{O}_M}$ and $ \lera{ N, \mathcal{O}_N}$ be top. sp. A map $ f: M \longrightarrow N$ is called \fbox{continuous} with respect to $ \mathcal{O}_M$ and $ \mathcal{O}_N$ iff

    \begin{align*}
        & V \in \mathcal{O}_N \implies \text{preim}_f(V) \in \mathcal{O}_M \\
    \end{align*}

    i.e. preimage of open sets are open.
\end{definition}

We have defined, for $ f: M \longrightarrow N$ and $ V \subseteq N$

\begin{align*}
    & \text{preim}_f(V) = \{ m \in M \mid f(m) \in V \} \\
\end{align*}

\begin{proposition}
    Let, $ \lera{ M, \mathcal{O}_M}$, $ \lera{ N, \mathcal{O}_N}$, $ \lera{ P, \mathcal{O}_P}$ be three top. sp.

    $ f: M \longrightarrow N$, $ g: N \longrightarrow P$ are continuous maps.

    The map $ h = g \circ f$ is also continuous w.r.t. $ \mathcal{O}_M$ \& $ \mathcal{O}_P$.
\end{proposition}

\begin{proof}
    Let, $ V \in \mathcal{O}_P$

    Now,

    \begin{align*}
        \text{preim}_h(V) &= \{ m \in M \mid h(m) \in V \} \\
        &= \{ m \in M \mid \lera{ g \circ f}(m) \in V \} \\
        &= \{ m \in M \mid f(m) \in \text{preim}_g(V) \} \\
        &= \text{preim}_f \lera{ \text{preim}_g(V)} \\
        &\in \mathcal{O}_M \\
    \end{align*}
\end{proof}

\section{Inheriting a topology}

Let $ \lera{ T, \mathcal{O}}$ be a top. sp. and $ S \subseteq T$.

\begin{align*}
    & \mathcal{O}|_S := \{ U \cap S \mid U \in \mathcal{O} \} \\
\end{align*}

$ \lera{ S, \mathcal{O}|_S}$ is a top. sp.

\begin{proof}
    \begin{align*}
        & \emptyset = \emptyset \cap S \text{ and } \emptyset \in \mathcal{O} \implies \emptyset \in \mathcal{O}|_S \\
        & S = S \cap M \text{ and } M \in \mathcal{O} \implies S \in \mathcal{O}|_S \\
    \end{align*}

    Let, $ S_1$ and $ S_2 \in \mathcal{O}|_S$

    \begin{align*}
        & \implies \exists U_1, U_2 \in \mathcal{O} \text{ such that } S_1 = U_1 \cap S, \; S_2 = U_2 \cap S \\
        & \implies U_1 \cap U_2 \in \mathcal{O} \implies \lera{ U_1 \cap U_2} \cap S \in \mathcal{O}|_S \\
        & \implies \lera{ U_1 \cap S} \cap \lera{ U_2 \cap S} \in \mathcal{O}|_S \implies S_1 \cap S_2 \in \mathcal{O}|_S \\
    \end{align*}

    Let, $ \alpha \in I$ and $ S_\alpha \in \mathcal{O}|_S$ $ \forall \alpha \in I$

    \begin{align*}
        & \implies \forall \alpha \; \exists U_\alpha \in \mathcal{O} \text{ such that } S_\alpha = U_\alpha \cap S \\
        & \implies \bigcup_{\alpha \in I} U_\alpha \in \mathcal{O} \implies \lera{ \bigcup_{\alpha \in I} U_\alpha} \cap S \in \mathcal{O}|_S \\
        & \implies \bigcup_{\alpha \in I} \lera{ U_\alpha \cap S} \in \mathcal{O}|_S \implies \bigcup_{\alpha \in I} S_\alpha \in \mathcal{O}|_S \\
    \end{align*}
\end{proof}

\section{Top. Manifold}

A Hausdorff, second countable top. space $ \lera{ M, \mathcal{O}}$ is called a d-dimensional topological manifold if for every point $ p \in M$, there is an open set U such that $ p \in U$ and there is a homeomorphism $ x: U \longrightarrow x(U) \subseteq \mathbb{R}^d$

where $ x(U) \in \mathcal{O}_{std}$

Note:- The map x is bi-continuous w.r.t. the top. spaces $ \lera{ U, \mathcal{O}|_U}$ and $ \lera{ x(U), \mathcal{O}_{std}|_{x(U)}}$

\begin{definition}
    Chart :=

    Let $ \lera{ M, \mathcal{O}}$ be a d-dim top. mfd. Then the pair $ \lera{ U, x}$ where $ U \in \mathcal{O}$ and x is a homeomorphism $ x: U \longrightarrow x(U) \subseteq \mathbb{R}^d$, is called a chart of the manifold.
\end{definition}

\begin{definition}
    Atlas :=

    An atlas of a manifold M is a collection of charts

    \begin{align*}
        & \mathcal{A} := \{ \lera{ U_\alpha, x_\alpha} \mid \alpha \in A \} \text{ such that} \\
        & \bigcup_{\alpha \in A} U_\alpha = M \\
    \end{align*}
\end{definition}

\section{Compatibility of charts}

Two charts $ \lera{ U,x}$ and $ \lera{ V,y}$ are $ \mathcal{C}$-compatible is either $ U \cap V \neq \emptyset$ are the maps

\begin{align*}
    & x \circ y^{-1} : y \lera{ U \cap V} \longrightarrow x \lera{ U \cap V} \\
    & y \circ x^{-1} : x \lera{ U \cap V} \longrightarrow y \lera{ U \cap V} \\
\end{align*}

is undergraduate-$ \mathcal{C}$

\begin{definition}
    $ \mathcal{C}$-Atlas :=

    An atlas $ \mathcal{A}$ of a top. manifold is called a $ \mathcal{C}$-Atlas if the charts of $ \mathcal{A}$ are $ \mathcal{C}$-compatible.
\end{definition}

\begin{definition}
    $ \mathcal{C}^k$-manifold:

    A $ \mathcal{C}^k$-manifold is a triple $ \lera{ M, \mathcal{O}, \mathcal{A}}$ where $ \lera{ N, \mathcal{O}}$ is a top. manifold and $ \mathcal{A}$ is a maximal $ \mathcal{C}^k$-atlas.
\end{definition}

\begin{definition}
    Smooth-manifold:

    A $ \mathcal{C}^\infty$-manifold is called a smooth manifold.
\end{definition}

From now on we are going to explicitly consider smooth manifolds.

\section{Maps between manifolds}

$ \phi: M \longrightarrow N$ is a map from M to N.

\section{Algebra of Forms}








At a point p in a d-dimensional manifold M, a special class of tensors are very important and deserves a seperate section of discussion. They are the totally antisymmetric (0,n)-tensors. To define them properly, let me first define an operator called the ``Antisymmetrization Operator''. For an (0,n)-tensor T at a point p in the manifold, and for a set of n linearly independent vectors $ \{ \overset{1}{X}, \overset{2}{X},...,\overset{n}{X}\}$ the action of the antisymmetrization operator $ \hat{A}$ on the tensor T is defined as,

\begin{align*}
    \hat{A}(T)\lera{\overset{1}{X}, \overset{2}{X},...,\overset{n}{X}} &= \frac{1}{n!} \sum_{ \pi} \delta_\pi \; T \lera{\overset{ \pi(1)}{X}, \overset{\pi(2)}{X},...,\overset{\pi(n)}{X}} \\
\end{align*}

where $ \pi$ denotes the different permutations of $ \{ 1,2,...,n \}$ and $ \tt{ \delta}{}{ \pi}$ is the sign of the permutation. 


The defintion of $ \hat{A}$ means, $ \tt{\hat{A} \lera{T}}{}{abc} = \tt{T}{}{\left[abc\right]}$, etc. 

An n-form is a (0,n)-tensor which is the same as its antisymmetrized version. So, if $ \hat{A} \lera{T}= T$ then T is an n-form at p. As such the components of T in any chart is totally antisymmetric, i.e. antisymmetric under the exchange of any two pairs of indices. $ \tt{T}{}{abc}= \tt{T}{}{[abc]}$.

It is obvious that there can not be an n-form where $ n > d$. In that case, at least two of the indices must be the same. For example if $ d=3 $, then a 4-form must have indices like $ \tt{T}{}{1233}$. However, being totally antisymmetric, $ \tt{T}{}{1233}$ must be equal to $ -\tt{T}{}{1233}$ and that is possible only if $ \tt{T}{}{1233}$ is 0. 

For a component of an n-form to be non-zero, all the indices must be different. So, for example, for a 2-form A in $ d= 3$, $ \tt{A}{}{11}= \tt{A}{}{22} = \tt{A}{}{33}= 0$, however, $ \tt{A}{}{12}, \tt{A}{}{13}, \tt{A}{}{23}$ can be non-zero. 

We define a scalar as a 0-form. The space of all n-forms at the point p is denoted by $ \Omega^n_p \lera{M}$. The space of all n-form fields is denoted by $ \Omega^n(M)$. An n-form field is also called a ``differential n-form''. 

















\subsection{Wedge Product}
Given a p-form A and a q-form B at the point p, we can make a $ (p+q)$-form
through an operation called wedge product. It is denoted by the symbol $ \wedge$. Let us define this in stages. First, for a set of n 1-forms $ \{\overset{1}{ \eta},..., \overset{n}{ \eta}\}$ their wedge product is defined as, 
\begin{align*}
    \overset{1}{ \eta}  \wedge ...\wedge \overset{n}{ \eta} = n! \; \overset{[1}{ \eta} \wedge ... \wedge \overset{n]}{ \eta} \\
\end{align*}

Suppose that at point p, we have a basis of $ T_p^* M$, $ \left\{E^1,...,E^d\right\}$. The wedge product between any n of the basis elements is given by, 

\begin{align*}
    \tt{E}{a_1}{} \wedge ...\wedge E^{a_n} = n! \; E^{[a_1} \otimes...\otimes E^{a_n]} \numberthis \label{basiswedge} \\
\end{align*}


This, in turn, gives us a new way to express any $ \omega \in \Omega^n_p(M)$ wrt the basis $ \{E^i\}$. 


 
\begin{align*}
    \omega &= \omega_{a_1...a_n} E^{a_1} \otimes ... \otimes E^{a_n} \\
 &= \omega_{a_1...a_n} E^{[a_1} \otimes ... \otimes E^{a_n]} \\
 & \text{The indices of $ \omega$ are all totally antisymmetric.} \\
 &= \frac{1}{n!} \omega_{a_1...a_n} \; n! \; E^{[a_1} \otimes ... \otimes E^{a_n]} \\
  & \text{So, using \cref{basiswedge}} \\
   \omega &= \frac{1}{n!}  \omega_{a_1...a_n}  E^{a_1} \wedge ... \wedge E^{a_n} \\
\end{align*}

People also write it a bit differently sometimes, summing over only increasing indices or decreasing indices. \textcolor{blue}{finish this thought.}


So now, given a p-form A and a q-form B, what are the components of $ A \wedge B$ with respect to the basis $ \left\{E^1,...,E^d\right\}$? 

\begin{align*}
    A \wedge B &= \frac{1}{p!} \frac{1}{q!} \tt{A}{}{a_1 ... a_{p}} \tt{B}{}{b_1 ... b_q} \tt{E}{a_1}{} \wedge ... \wedge \tt{E}{a_p}{} \wedge \tt{E}{b_1}{} \wedge ... \tt{E}{b_q}{} \\
    &= \frac{ \lera{p+q}!}{p! q!}  \tt{A}{}{a_1 ... a_{p}} \tt{B}{}{b_1 ... b_q} \tt{E}{[a_1}{} \otimes ... \otimes \tt{E}{b_q]}{} \\
    &= \frac{ \lera{p+q}!}{p! q!}  \tt{A}{}{[a_1 ... a_{p}} \tt{B}{}{b_1 ... b_q]} \tt{E}{a_1}{} \otimes ... \otimes \tt{E}{b_q}{} \\
\end{align*}


So,

\begin{align*}
    \lera{A \wedge B}_{[a...bc...d]} &= \frac{ \lera{p+q}!}{p! q!} \tt{A}{}{[a...b} \tt{B}{}{c...d]} \numberthis \label{wedge_component}\\
\end{align*}

For two 1-forms $ \omega$ and $ \sigma$ we know from the definition of one wedge products that $ \omega \wedge \sigma = \omega \otimes \sigma - \sigma \otimes \omega$. This also means $ \omega \wedge \sigma = - \sigma \wedge \omega $. The rule is, anytime we swap two one forms, there is a minus sign. Remembering this rule will lead us to a graded commutation rule for the wedge product of a p-form with a q-form. It is called graded, because the anti-commutation rule depends on the degree of the forms involved. Here is the rule (assuming that $ \omega, \sigma$ are p and q forms, respectively). 

\begin{align*}
     \sigma \wedge \omega &= \lera{ -1}^{pq} \omega \wedge \sigma \numberthis \label{gradedcommutation} \\
\end{align*}

To see this, let's write the forms in a basis. So, $ \omega = \frac{1}{p!} \omega_{a...b}  \tt{E}{a}{} \wedge ... \wedge \tt{E}{b}{}$ and $ \sigma = \frac{1}{q!} \sigma_{c...d}  \tt{E}{c}{} \wedge ... \wedge \tt{E}{d}{} $. Now, $ \sigma \wedge \omega = \frac{1}{p! q! } \tt{ \sigma}{}{a...b} \tt{ \omega}{}{c...d} \tt{E}{a}{} \wedge...\wedge \tt{E}{b}{} \wedge \tt{E}{c}{} \wedge ... \wedge \tt{E}{d}{}$. To go to $ \omega \wedge \sigma$, the trick is to take the basis vectors from the second set $ E^{c},...,E^{d}$ one by one and take them p steps leftward. Each time I do this, I get a $ \lera{-1}^p$, and I have to do this for q members of the second set, so all in all, I have $ \lera{-1}^{pq}$. And in the end, the basis vectors will look like, $  \lera{ -1}^{pq}\tt{E}{c}{} \wedge...\wedge \tt{E}{d}{} \wedge \tt{E}{a}{} \wedge ... \wedge \tt{E}{b}{}$. So, 

\begin{align*}
    \sigma \wedge \omega &=  \frac{1}{p! q! } \tt{ \sigma}{}{a...b} \tt{ \omega}{}{c...d} \tt{E}{a}{} \wedge...\wedge \tt{E}{b}{} \wedge \tt{E}{c}{} \wedge ... \wedge \tt{E}{d}{}\\
    &= \lera{-1}^{pq}\frac{1}{p! q! } \tt{ \sigma}{}{a...b} \tt{ \omega}{}{c...d} \tt{E}{c}{} \wedge...\wedge \tt{E}{d}{} \wedge \tt{E}{a}{} \wedge ... \wedge \tt{E}{b}{}\\
    &= \lera{-1}^{pq}\frac{1}{p! q! }  \tt{ \omega}{}{c...d}\tt{ \sigma}{}{a...b} \tt{E}{c}{} \wedge...\wedge \tt{E}{d}{} \wedge \tt{E}{a}{} \wedge ... \wedge \tt{E}{b}{}\\
    &= \lera{ -1}^{pq} \omega \wedge \sigma \\
\end{align*}





















\subsection{Exterior Derivative}

I will define this using coordinate basis. For derivatives, we need n-form fields as we can not do a derivation to a quantity which is defined at a point p. So we will work in a chart $ \lera{U,x}$. The exterior derivative of a 0-form, i.e. a function f is defined as, $ df := \partial_a f dx^a $. Notice that the exterior derivative of a zero form gives a 1-form. This is a general feature of exterior derivatives as we will see below that the exterior derivative of a p-form is a $ \lera{p+1}$-form. Now, given how a function's exterior derivative is taken, we define the exterior derivative of a p-form $ A = \frac{1}{p!} \tt{A}{}{a_1 ...a_p} dx^{a_1} \wedge ... \wedge dx^{a_p}$ to be, 

\begin{align*}
    dA := \frac{1}{p!} d(A_{a_1...a_p}) \wedge \tt{dx}{a_1}{} \wedge ...\wedge \tt{dx}{a_p}{} \numberthis \label{exterior_derivative_defined} \\
\end{align*}


To see the components of dA, we expand the RHS of \cref{exterior_derivative_defined}. $ dA = \frac{1}{p!} \partial_m A_{a_1...a_p} dx^m \wedge dx^{a_1} \wedge ...dx^{a_p} = \frac{ \lera{p+1}!}{p!} \partial_m \tt{A}{}{a_1...a_p} dx^{[m} \otimes dx^{a_1} \otimes...\otimes dx^{a_p]} = \lera{p+1}\partial_{[m} \tt{A}{}{a_1...a_p]} dx^{m} \otimes dx^{a_1} \otimes...\otimes dx^{a_p}$ . So the $ \tt{dA}{}{ma_1...a_p}$ component of $ dA$ is given by $ \tt{dA}{}{ma_1...a_p}= \lera{p+1} \partial_{[m} \tt{A}{}{a_1...a_p]} $ .


We reiterate here that the RHS of \cref{exterior_derivative_defined} is a tensor by definition ( exterior derivative of a function is a 1-form and wedge product of differential 1-forms is also a differential form).It is quite fascinating that a fully covariant notion of derivative can be cooked up from using only coordiante based partial derivative.

\section{Non-coordinate Basis and Tetrad Formalism}

For non-coordinate basis, I will use Greek indices and for coordinate basis I will use Latin indices. 


Here is a consistent story, 
\begin{align*}
    g \lera{ \tt{E}{}{ \mu}, \tt{E}{}{ \nu}} &= \tt{ \eta}{}{ \mu \nu } \numberthis \label{ tetrad}
\end{align*}

Since $ \tt{E}{ \mu}{}$ is a covecotor, it can be expanded in terms of the coordinate basis, 
\begin{align*}
    \tt{E}{ \mu}{} &= \tt{E}{ \mu} {a} dx^ {a} \\
\end{align*}

Now, E can be thought of as a matrix. Our convention is as usual, upper index is for row, lower index is for column. And, for any matrix A, $ \tt{ \lera{A^{-1}}}{i}{j}$ is denoted by $A_j^{\;\;i}$. Based on this, $ dx^a$ can be expanded as, 

\begin{align*}
    dx^{a} &= E_{ \mu}^{\;\;a} \tt{E}{ \mu}{}
\end{align*}

Now, the non-coordinate basis vectors $ \tt{E}{}{ \mu}$ are dual to $ \tt{E}{ \mu}{}$, that is, $ \tt{E}{ \mu}{} \lera{ \tt{E}{}{ \nu}}= \kron{ \mu}{ \nu}$. The basis vectors $ \tt{E}{}{ \mu}$ can also be expanded in terms of the coordiante basis vectors $ \partial_a$, as, $\tt{E}{}{ \mu}= F_{ \mu}^{ \;\; a} \partial_a $. The nice thing is that F is nothing but E! Let me show that.

\begin{align*}
    &\tt{E}{ \mu}{} \lera{ \tt{E}{}{ \nu}} = \kron{ \mu}{ \nu} \\
    & \tt{E}{ \mu }{a} dx^a \lera{ F_{ \nu}^{\;\;b} \partial_b} = \kron{ \mu}{ \nu} \\
    & \tt{E}{ \mu}{a } F_{ \nu}^{\;\;b} dx^a \lera{ \partial_b} = \kron{ \mu}{ \nu} \\
    & \tt{E}{ \mu}{a } F_{ \nu}^{\;\;b} \kron{a}{b} = \kron{ \mu}{ \nu} \\
    &  \tt{E}{ \mu}{a } F_{ \nu}^{\;\;a}  = \kron{ \mu}{ \nu} \\
    & \tt{E}{ \mu}{ a} \tt{ \lera{F^{-1}}}{a}{ \nu}= \kron{ \mu}{ \nu} \\
\end{align*}

This simply means, $ F^{-1} = E^{-1}$ or, $ F=E$. And,
\begin{align*}
    \tt{E}{}{ \mu}=& E_{ \mu}^{ \;\; a} \partial_a
\end{align*}

\begin{align*}

    & \partial_b = \tt{E}{ \mu}{b} \tt{E}{}{ \mu} \\

    =&\tt{E}{ \nu}{} \lera{ \cov{ \tt{E}{}{ \mu}}{ \lera{ \pp{ x^m}{x'^a} \partial_m}}}\\
    =&\tt{E}{ \nu}{} \lera{ \pp{ x^m}{x'^a}\cov{ \tt{E}{}{ \mu}}{  \partial_m}}\\ 
    =& \pp{ x^m}{x'^a}  \tt{E}{ \nu}{}\lera{ \cov{ \tt{E}{}{ \mu}}{ \partial_m}}\\
    =& \pp{ x^m}{x'^a} \omega_\mu{}^\nu{}_m
\end{align*}

This proves that $ \omega_\mu{}^\nu{}_a$ is a 1-form in the index a. To show that it is NOT a tensor in the Greek indices, I now change only the frame and not the chart, then,

\begin{align*}
    \omega_{\mu'}{}^{ \nu'}{}_a &= \tt{E'}{ \nu}{} \lera{ \cov{ \tt{E'}{}{ \mu}}{ \partial_a}} \\
    &= \tt{ \Lambda}{ \nu}{ \alpha} \tt{E}{ \alpha}{} \lera{ \cov{ \lera{ \Lambda_ \mu{}^ \beta E_ \beta}}{ \partial_a}} \\
    &= \tt{ \Lambda}{ \nu}{ \alpha} \tt{E}{ \alpha}{} \lera{ \Lambda_ \mu{^ \beta }}
\end{align*}


\textcolor{blue}{Write down expressions for $ \cov{ \tt{A}{ \mu}{}}{\nu}$ just like you know how to write $ \cov{ A^a}{ m}$ etc.}

One immediate inquiry is the expression of $ \dud{ \omega}{ \mu}{ \nu }{a}$ in terms of $ \cris{a}{bc}$. We know that, 

\begin{align*}
    &\cov{ \tt{A}{ \mu}{}}{ a} = \cov{ A^b}{a} \tt{E}{ \mu}{b} \\
    & \partial_a \lera{ \tt{A}{ \mu}{}} + \dud{ \omega}{ \rho}{ \mu}{a} \tt{A}{ \rho}{} = \lera{ \partial_a \tt{A}{b}{} + \cris{b}{ma} \tt{A}{m}{}} \tt{E}{ \mu }{b} \\
    & \partial_a \lera{ \tt{E}{ \mu}{m} A^m} + \dud{ \omega}{ \rho}{ \mu}{a} \tt{A}{ \rho}{} = \lera{ \partial_a \tt{A}{b}{} + \cris{b}{ma} \tt{A}{m}{}} \tt{E}{ \mu }{b} \\
    & \partial_a \lera{ \tt{E}{ \mu}{m} A^m} + \dud{ \omega}{ \rho}{ \mu}{a} \tt{E}{ \rho}{m}\tt{A}{ m}{} = \lera{ \partial_a \tt{A}{b}{} + \cris{b}{ma} \tt{A}{m}{}} \tt{E}{ \mu }{b} \\
    & \tt{A}{ m}{} \partial_a \tt{E}{ \mu}{m} + \tt{E}{ \mu}{m} \partial_a \tt{A}{m}{} + \dud{ \omega}{ \rho}{ \mu}{a} \tt{E}{ \rho}{m}\tt{A}{ m}{} = \tt{E}{ \mu}{m } \partial_a \tt{A}{m}{} + \cris{n}{ ma} \tt{E}{ \mu}{n} \tt{A}{m}{}\\
    & \tt{A}{ m}{} \partial_a \tt{E}{ \mu}{m} + \dud{ \omega}{ \rho}{ \mu}{a} \tt{E}{ \rho}{m}\tt{A}{ m}{} =  \cris{n}{ ma} \tt{E}{ \mu}{n} \tt{A}{m}{}\\
    &  \partial_a \tt{E}{ \mu}{m} + \dud{ \omega}{ \rho}{ \mu}{a} \tt{E}{ \rho}{m} =  \cris{n}{ ma} \tt{E}{ \mu}{n} \\
    & \text{Now simply multiply by $ \du{E}{ \nu}{m}$ on both sides.}\\
    &\du{E}{ \nu}{m}\partial_a \tt{E}{ \mu}{m} + \dud{ \omega}{ \rho}{ \mu}{a} \tt{E}{ \rho}{m}\du{E}{ \nu}{m} =  \cris{n}{ ma} \tt{E}{ \mu}{n} \du{E}{ \nu}{m}\\
    &\du{E}{ \nu}{m}\partial_a \tt{E}{ \mu}{m} + \dud{ \omega}{ \rho}{ \mu}{a} \kron{ \rho}{ \nu} =  \cris{n}{ ma} \tt{E}{ \mu}{n} \du{E}{ \nu}{m}\\
    & \dud{ \omega}{ \nu}{ \mu}{a} = \cris{n}{ma} \tt{E}{ \mu}{n} \du{E}{ \nu}{m} - \partial_a \tt{E}{ \mu}{m} \;\; \du{ E}{ \nu}{m} \numberthis \label{ omegaintermsofcris}\\
\end{align*}


Similarly, multiplying both sides of \cref{ omegaintermsofcris} with $ \du{E}{ \mu}{p} \tt{E}{ \nu}{q}$ so that the two E's with the Christoffel symbol becomes Kronecker deltas and ultimately go away, 

\begin{align*}
    &\dud{ \omega}{ \nu}{ \mu}{a } \du{E}{ \mu}{p} \tt{E}{ \nu}{q} = \cris{n}{ma} \du{E}{ \mu}{p } \tt{E}{ \mu}{n} \du{E}{ \nu}{m} \tt{E}{ \nu}{q} - \partial_a \tt{E}{ \mu}{m} \; \du{E}{ \nu}{m} \tt{E}{ \nu}{q} \du{E}{ \mu}{p} \\
    & \dud{ \omega}{ \nu}{ \mu}{a } \du{E}{ \mu}{p} \tt{E}{ \nu}{q} = \cris{n}{ma} \kron{p}{n} \kron{m}{q} - \partial_a \tt{E}{ \mu}{m} \kron{m}{q} \du{E}{ \mu}{p} \\
    & \dud{ \omega}{ \nu}{ \mu}{a } \du{E}{ \mu}{p} \tt{E}{ \nu}{q}= \cris{p}{qa} - \partial_a \tt{E}{ \mu}{q} \; \du{E}{ \mu}{p} \\
    & \cris{p}{qa} = \dud{ \omega}{ \nu}{ \mu}{a } \du{E}{ \mu}{p} \tt{E}{ \nu}{q} + \partial_a \tt{E}{ \mu}{q} \; \du{E}{ \mu}{p} \\
    & \text{Just replace p with m, q with n, and swap $ \mu$ and $ \nu$} \\
    & \cris{m}{na} = \dud{ \omega}{ \mu}{ \nu}{a} \du{E}{ \nu}{ m} \tt{E}{ \mu}{n} + \partial_a \tt{E}{ \nu}{n} \; \du{E}{ \nu}{m} \numberthis \label{crisintermsofomega} \\
\end{align*}


This leaves us to defining another quantity in this section, called the Ricci rotation coefficients. It is basically the connection one form expressed in the frame basis instead of coordinate basis. That is, while a connection one form $ \dud{ \omega}{ \mu}{ \nu}{a}$ has the 1-form index ``a'' from the frame basis, the Ricci rotation coefficient $ \dud{ \omega}{ \mu}{ \nu}{ \rho}$ is the connection 1 form expanded in terms of $ \{ \tt{E}{ \mu}{}\}$. 

\begin{align*}
    &\dud{ \omega}{ \mu}{ \nu}{ \rho} = \dud{ \omega}{ \mu}{ \nu}{ a} \du{E}{ \rho}{a} \numberthis \label{riccirotationdefined}
\end{align*}

Another way to express the Ricci rotation coefficient is, $ \dud{ \omega}{ \mu}{ \nu}{ \rho} = \tt{E}{ \nu}{} \lera{ \cov{ \tt{E}{}{ \mu}}{ \tt{E}{}{ \rho}}}$. We can quickly see that as follows, by starting with the RHS,

\begin{align*}
    \tt{E}{ \nu}{} \lera{ \cov{ \tt{E}{}{ \mu}}{ \tt{E}{}{ \rho}}} &= \tt{E}{ \nu}{} \lera{ \cov{ \tt{E}{}{ \mu}}{ \lera{ \du{E}{ \rho}{m} \partial_m}}} \\
    &= \tt{E}{ \nu}{} \lera{ \du{E}{ \rho}{m} \cov{ \tt{E}{}{ \mu}}{ \partial_m}} \\
    &=  \du{E}{ \rho}{m}\tt{E}{ \nu}{} \lera{ \cov{ \tt{E}{}{ \mu}}{ \partial_m}} \\
    &= \du{E}{ \rho}{m} \dud{ \omega}{ \mu}{ \nu}{m} \\
    &= \dud{ \omega}{ \mu}{ \nu}{ \rho} \\
\end{align*}

\subsubsection{Torsion and Curvature 2-forms}
I will first simply define the Torsion and Curvature 2-forms, then show that these definitions work. There are other ways of arriving at these definitions that might be more direct, but since we have to memorize these anyways, I would simply state the definition without motivation. 

The torsion 2-form is defined as (suppressing the latin indices),

\begin{align*}
    \tt{T}{ \mu }{} = d \tt{E}{ \mu}{} + \du{ \omega}{ \nu}{ \mu} \wedge \tt{E}{ \nu}{}  \numberthis \label{torsionform}\\
\end{align*}

The curvature 2-form is defined as (again, supressing the latin indices),

\begin{align*}
    \tt{R}{ \mu}{ \nu} = d \du{ \omega}{ \nu}{ \mu} + \du{ \omega}{ \alpha}{ \mu} \wedge \du{ \omega}{ \nu}{ \alpha}  \numberthis \label{curvatureform}\\
\end{align*}



To show that the definition of torsion form works, start with expanding the RHS of \cref{torsionform},

\begin{align*}
    d \tt{E}{ \mu}{ab} + \left( \du{ \omega}{ \nu}{ \mu} \wedge \tt{E}{ \nu } \right)_{ab} &= \partial_a \tt{E}{ \mu}{b} - \partial_b \tt{E}{ \mu}{a} + \dud{ \omega }{ \nu}{ \mu}{a } \tt{E}{ \nu}{b}-\dud{ \omega }{ \nu}{ \mu}{b } \tt{E}{ \nu}{a} \numberthis \label{torsionform_rhs_expanded} \\
\end{align*}


Now, \begin{align*}
    \tt{T }{ \mu}{ ab} &= T \lera{ \tt{E}{ \mu}{}, \partial_a, \partial_b} \\
    &= \tt{E}{ \mu}{} \lera{ \cov{ \partial_b}{a}- \cov{ \partial_a}{b}} \\
    &= \tt{E}{ \mu}{} \left(\cov{ \partial_b}{a} \right) - \lera{ a \leftrightarrow b}\\
    &= \tt{E}{ \mu}{} \lera{ \cris{ m}{ba} \partial_m} -\lera{ a \leftrightarrow b} \\
    &= \tt{E}{ \mu}{ } \lera{ \cris{m}{ba} \tt{E}{ \nu}{m} \tt{E}{}{ \nu}} - \lera{ a \leftrightarrow b}\\
    &=\cris{m}{ba} \tt{E}{ \nu}{m} \kron{ \mu}{ \nu} - \lera{ a \leftrightarrow b}\\
    &=\cris{m}{ba} \tt{E}{ \mu}{m} - \lera{ a \leftrightarrow b}\\
    &= \tt{E}{ \mu}{m } \lera{ \dud{ \omega}{ \alpha}{ \beta}{a} \du{E}{ \beta}{ m} \tt{E}{ \alpha}{b} + \partial_a \tt{E}{ \beta}{b} \; \du{E}{ \beta}{m}} - \lera{ a \leftrightarrow b}\\ 
    &= \dud{ \omega}{ \alpha}{ \beta}{a} \tt{E}{ \mu}{m} \du{E}{ \beta}{m } \tt{E}{ \alpha}{b} + \partial_a \tt{E}{ \beta}{b} \;\tt{E}{ \mu}{m} \du{E}{ \beta}{m} - \lera{ a \leftrightarrow b}\\
    &= \dud{ \omega}{ \alpha}{ \beta}{a} \kron{ \mu}{ \beta}\tt{E}{ \alpha}{b} + \partial_a \tt{E}{ \beta}{b} \; \kron{ \mu}{ \beta} - \lera{ a \leftrightarrow b}\\
    &= \partial_a \tt{E}{ \mu}{b} +\dud{ \omega}{ \alpha}{ \mu}{a} \tt{E}{ \alpha}{b} - \lera{ a \leftrightarrow b}\\
    &= \partial_a \tt{E}{ \mu}{b} +\dud{ \omega}{ \nu}{ \mu}{a} \tt{E}{ \nu}{b} -\partial_b \tt{E}{ \mu}{a} -\dud{ \omega}{ \nu}{ \mu}{b} \tt{E}{ \nu}{a} \\
    &=\partial_a \tt{E}{ \mu}{b}-\partial_b \tt{E}{ \mu}{a}+\dud{ \omega}{ \nu}{ \mu}{a} \tt{E}{ \nu}{b}-\dud{ \omega}{ \nu}{ \mu}{b} \tt{E}{ \nu}{a}\\
    & \text{This matches with \cref{torsionform_rhs_expanded}}
\end{align*}
